38 public static void Initialize()
39 {
40 Directory.CreateDirectory("Plugins");
41 foreach (string s in Directory.GetFiles("Plugins"))
42 {
43 if (!s.EndsWith(".dll") || Plugin.Plugins.ContainsKey(Path.GetFileName(s)))
44 continue;
45 // Load the plugin assembly
46 Assembly pluginAssembly = Assembly.LoadFile(Environment.CurrentDirectory + "/" + s);
47 // Find the type which implements the IPlugin interface
48 var pluginType = pluginAssembly.GetTypes().FirstOrDefault(t => typeof(IPlugin).IsAssignableFrom(t) && !t.IsInterface);
49 if (pluginType != null)
50 {
51 // Create an instance of this type
52 IPlugin pluginInstance = Activator.CreateInstance(pluginType) as IPlugin;
53 Plugin.Plugins.Add(Path.GetFileName(s), pluginInstance);
54 if (pluginInstance.ContextMenu)
55 App.AddContextMenu(pluginInstance.MenuPath);
56 else
57 App.AddMenu(pluginInstance.MenuPath);
58 }
59 else
60 {
61 Console.WriteLine("No plugin found in the assembly: " + s);
62 }
63 }
