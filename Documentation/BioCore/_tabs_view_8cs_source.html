96 public void AddTab(BioImage b)
97 {
98 if (b == null)
99 throw new ArgumentNullException(nameof(b), "The image loader returned no image to display.");
100 if (b.filename.Contains('/') || b.filename.Contains('\\'))
101 b.filename = Path.GetFileName(b.filename);
102 TabPage t = new TabPage(GetAvailableTabTitle(b.filename));
103 ImageView v = new ImageView(b);
104 v.Dock = DockStyle.Fill;
105 t.Controls.Add(v);

107 {
108 Width = b.SizeX;

110 }
111 tabControl.TabPages.Add(t);
112 tabControl.Dock = DockStyle.Fill;
113 tabControl.SelectedIndex = tabControl.TabCount - 1;
114 viewers.Add(v);
115 ResizeView();
116 if (!Visible)
117 Show();
118 if (WindowState == FormWindowState.Minimized)
119 WindowState = FormWindowState.Normal;
120 Activate();
121 }

123 private string GetAvailableTabTitle(string filename)
124 {
125 var titles = new HashSet<string>(tabControl.TabPages.Cast<TabPage>().Select(page => page.Text), StringComparer.OrdinalIgnoreCase);
126 if (!titles.Contains(filename))
127 return filename;
128
129 // Keep the compound OME extension together when numbering copies.
130 string extension = filename.EndsWith(".ome.zarr", StringComparison.OrdinalIgnoreCase) ? filename.Substring(filename.Length - 9)
131 : filename.EndsWith(".ome.tiff", StringComparison.OrdinalIgnoreCase) ? filename.Substring(filename.Length - 9)
132 : filename.EndsWith(".ome.tif", StringComparison.OrdinalIgnoreCase) ? filename.Substring(filename.Length - 8)
133 : Path.GetExtension(filename);
134 string stem = filename.Substring(0, filename.Length - extension.Length);
135 for (int number = 2; ; number++)
136 {
137 string title = $"{stem}-{number}{extension}";
138 if (!titles.Contains(title))
139 return title;
140 }
141 }

252 private async void MenuItem_Click(object? sender, EventArgs e)
253 {
254 BioImage source = Image;
255 if (source == null) return;
256 ToolStripMenuItem m = (ToolStripMenuItem)sender;
257 string command = m.Tag as string ?? m.Text;
258 string escaped = command.Replace("\\", "\\\\").Replace("\"", "\\\"");
259 try
260 {
261 BioImage result = await Fiji.RunOnImage(source, "run(\"" + escaped + "\");", BioConsole.headless, BioConsole.onTab, BioConsole.useBioformats, BioConsole.newTab);
262 if (result == null)
263 {
264 MessageBox.Show(this, $"'{command}' did not return an image. It may have produced another type of result or failed in Fiji.",
265 "Fiji Command", MessageBoxButtons.OK, MessageBoxIcon.Information);
266 return;
267 }
268 if (ReferenceEquals(result, source))
269 {
270 ImageView? sourceViewer = viewers.FirstOrDefault(viewer => viewer.Images.Contains(source));
271 if (sourceViewer != null && !sourceViewer.IsDisposed)
272 {
273 TabPage? sourceTab = tabControl.TabPages.Cast<TabPage>()
274 .FirstOrDefault(page => page.Controls.Contains(sourceViewer));
275 if (sourceTab != null)
276 tabControl.SelectedTab = sourceTab;
277 App.viewer = sourceViewer;
278 sourceViewer.SelectedIndex = sourceViewer.Images.IndexOf(source);
279 sourceViewer.UpdateView();
280 }
281 }
282 else
283 AddTab(result);
284 }
285 catch (Exception ex)
286 {
287 Trace.WriteLine(ex.ToString());
288 MessageBox.Show(this, $"Could not run '{command}':\n\n{ex}", "Fiji Command", MessageBoxButtons.OK, MessageBoxIcon.Error);
289 return;
290 }
291 ToolStripMenuItem mi = new ToolStripMenuItem(m.Text);
292 mi.Tag = command;
293 mi.Click += MenuItem_Click;
294 bool con = false;
295 foreach (ToolStripMenuItem item in recentToolStripMenuItem.DropDownItems)
296 {
297 if ((item.Tag as string ?? item.Text) == command)
298 con = true;
299 }
300 if (!con)
301 recentToolStripMenuItem.DropDownItems.Add(mi);

1009 private async void ItemClicked(object sender, EventArgs e)
1010 {
1011 ToolStripMenuItem ts = (ToolStripMenuItem)sender;
1012 if (IsZarrSource(ts.Text))
1013 {
1014 await OpenZarrSourceAsync(ts.Text);
1015 return;
1016 }
1017 if (!BioImage.isOME(ts.Text))
1018 {
1019 BioImage[] bs = BioImage.OpenSeries(ts.Text, false);
1020 for (int i = 0; i < bs.Length; i++)
1021 {
1022 if (i == 0)
1023 AddTab(bs[i]);
1024 else
1025 Viewer.AddImage(bs[i]);
1026 }
1027 }
1028 else
1029 {
1030 OpenInTab tb = new OpenInTab();
1031 bool separateTabs = tb.ShowDialog() == DialogResult.Yes;
1032 bool firstImage = true;
1033 await foreach (BioImage image in ImageSeriesLoader.OpenAsync(ts.Text))
1034 {
1035 if (separateTabs || firstImage)
1036 AddTab(image);
1037 else
1038 Viewer.AddImage(image);
1039 firstImage = false;
1040 }
1041 }
1042 App.viewer.GoToImage();
1043 SaveProperties();
1044 }
