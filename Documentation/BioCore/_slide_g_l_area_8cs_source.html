67 private const string VertexShaderSource = @"
68#version 330 core
69
70layout(location=0) in vec2 aPos;
71layout(location=1) in vec2 aUV;
72
73uniform vec2 pos; // pixel-space top-left of tile
74uniform vec2 size; // pixel-space size of tile
75uniform vec2 viewportSize; // (width, height) in pixels
76
77out vec2 uv;
78
79void main()
80{
81 // aPos is in [0,1] x [0,1]
82 vec2 pixelPos = aPos * size + pos;
83
84 // Convert from pixel coords to NDC (-1..1), with Y flipped for GL
85 vec2 ndc;
86 ndc.x = (pixelPos.x / viewportSize.x) * 2.0 - 1.0;
87 ndc.y = 1.0 - (pixelPos.y / viewportSize.y) * 2.0;
88
89 gl_Position = vec4(ndc, 0.0, 1.0);
90 uv = aUV;
91}
92";
93
94 private const string FragmentShaderSource = @"
95#version 330 core
96
97in vec2 uv;
98out vec4 FragColor;
99uniform sampler2D tex;
100
101void main()
102{
103 // Image data is usually Top-Down. OpenGL Textures are Bottom-Up.
104 // This implicit flip means UV(0,0) (Top-Left Quad) maps to Texture Bottom (Image Top).
105 // This is the correct orientation for Top-Left origin rendering.
106 FragColor = texture(tex, vec2(uv.x, uv.y));
107}
108";

163 private void InitializeGL()
164 {
165 if (_glInitialized) return;
166
167 try
168 {
169 // Compile and link shader program
170 int vertexShader = CompileShader(ShaderType.VertexShader, VertexShaderSource);
171 int fragmentShader = CompileShader(ShaderType.FragmentShader, FragmentShaderSource);
172
173 _shaderProgram = GL.CreateProgram();
174 GL.AttachShader(_shaderProgram, vertexShader);
175 GL.AttachShader(_shaderProgram, fragmentShader);
176 GL.LinkProgram(_shaderProgram);
177
178 GL.GetProgram(_shaderProgram, GetProgramParameterName.LinkStatus, out int linkStatus);
179 if (linkStatus == 0)
180 {
181 string infoLog = GL.GetProgramInfoLog(_shaderProgram);
182 Console.WriteLine($"[SlideGLArea] Shader link FAILED: {infoLog}");
183 return;
184 }
185
186 GL.DeleteShader(vertexShader);
187 GL.DeleteShader(fragmentShader);
188
189 // Cache uniform locations
190 _locPos = GL.GetUniformLocation(_shaderProgram, "pos");
191 _locSize = GL.GetUniformLocation(_shaderProgram, "size");
192 _locViewportSize = GL.GetUniformLocation(_shaderProgram, "viewportSize");
193 _locTex = GL.GetUniformLocation(_shaderProgram, "tex");
194
195 Console.WriteLine($"[SlideGLArea] Uniform locations: pos={_locPos} size={_locSize} viewport={_locViewportSize} tex={_locTex}");
196
197 // Build the unit-quad VAO/VBO shared by all tile draws
198 float[] quadVertices =
199 {
200 // pos uv
201 0, 0, 0, 0,
202 1, 0, 1, 0,
203 1, 1, 1, 1,
204
205 0, 0, 0, 0,
206 1, 1, 1, 1,
207 0, 1, 0, 1
208 };
209
210 _vao = GL.GenVertexArray();
211 _vbo = GL.GenBuffer();
212
213 GL.BindVertexArray(_vao);
214 GL.BindBuffer(BufferTarget.ArrayBuffer, _vbo);
215 GL.BufferData(BufferTarget.ArrayBuffer, quadVertices.Length * sizeof(float),
216 quadVertices, BufferUsageHint.StaticDraw);
217
218 // Position attribute (location = 0)
219 GL.VertexAttribPointer(0, 2, VertexAttribPointerType.Float, false, 4 * sizeof(float), 0);
220 GL.EnableVertexAttribArray(0);
221
222 // UV attribute (location = 1)
223 GL.VertexAttribPointer(1, 2, VertexAttribPointerType.Float, false, 4 * sizeof(float), 2 * sizeof(float));
224 GL.EnableVertexAttribArray(1);
225
226 GL.BindVertexArray(0);
227
228 // Verify no GL errors accumulated during setup
229 var err = GL.GetError();
230 if (err != ErrorCode.NoError)
231 Console.WriteLine($"[SlideGLArea] GL error after init: {err}");
232
233 _glInitialized = true;
234 Console.WriteLine($"[SlideGLArea] GL initialization complete. VAO={_vao} VBO={_vbo} Program={_shaderProgram}");
235 }
236 catch (Exception ex)
237 {
238 Console.WriteLine($"[SlideGLArea] GL initialization FAILED: {ex.Message}");
239 }
240 }

288 private void RenderFrame()
289 {
290 int width = Width;
291 int height = Height;
292 if (width <= 0 || height <= 0) return;
293
294 // Handle High-DPI: viewport is in physical pixels, uniforms in logical pixels
295 double scale = DeviceDpi / 96.0;
296 GL.Viewport(0, 0, (int)(width * scale), (int)(height * scale));
297 GL.ClearColor(0f, 0f, 0f, 1f);
298 GL.Clear(ClearBufferMask.ColorBufferBit);
299
300 if (TilesToRender.Count == 0)
301 return;
302
303 GL.Enable(EnableCap.Blend);
304 GL.BlendFunc(BlendingFactor.SrcAlpha, BlendingFactor.OneMinusSrcAlpha);
305
306 GL.UseProgram(_shaderProgram);
307 GL.BindVertexArray(_vao);
308
309 // Viewport size uniform is constant for all tiles this frame
310 GL.Uniform2(_locViewportSize, (float)width, (float)height);
311 GL.Uniform1(_locTex, 0);
312 GL.ActiveTexture(TextureUnit.Texture0);
313
314 int rendered = 0;
315 int skipped = 0;
316 foreach (var tile in TilesToRender)
317 {
318 if (!_textureCache.TryGetValue(tile.Index, out int texId))
319 {
320 skipped++;
321 continue;
322 }
323
324 GL.Uniform2(_locPos, tile.ScreenX, tile.ScreenY);
325 GL.Uniform2(_locSize, tile.ScreenWidth, tile.ScreenHeight);
326
327 GL.BindTexture(TextureTarget.Texture2D, texId);
328 GL.DrawArrays(PrimitiveType.Triangles, 0, 6);
329 rendered++;
330 }
331
332 // Diagnostic: log once when tiles first appear (or fail to appear)
333 if (rendered == 0 && skipped > 0)
334 Console.WriteLine($"[SlideGLArea] RenderFrame: {TilesToRender.Count} tiles to render, but ALL {skipped} skipped (no texture in cache).");
335 else if (rendered > 0)
336 Console.WriteLine($"[SlideGLArea] RenderFrame: rendered {rendered} tiles, skipped {skipped}.");
337
338 GL.BindVertexArray(0);
339 GL.UseProgram(0);
340 GL.Disable(EnableCap.Blend);
341 }

351 public bool UploadTileTexture(TileIndex index, byte[] pixelData, int tileWidth, int tileHeight)
352 {
353 if (pixelData == null || pixelData.Length == 0)
354 return false;
355
356 // Guard: don't attempt GL operations if the context isn't ready
357 if (!IsHandleCreated)
358 {
359 Console.WriteLine($"[SlideGLArea] UploadTileTexture skipped — control handle not created yet.");
360 return false;
361 }
362
363 try
364 {
365 MakeCurrent();
366 }
367 catch (Exception ex)
368 {
369 Console.WriteLine($"[SlideGLArea] MakeCurrent failed in UploadTileTexture: {ex.Message}");
370 return false;
371 }
372
373 if (!_glInitialized)
374 InitializeGL();
375
376 if (!_glInitialized)
377 {
378 Console.WriteLine($"[SlideGLArea] UploadTileTexture skipped — GL init failed.");
379 return false;
380 }
381
382 // Already cached — nothing to do
383 if (_textureCache.ContainsKey(index))
384 return true;
385
386 // Evict old textures if cache is full
387 if (_textureCache.Count >= MAX_CACHED_TEXTURES)
388 EvictOldestTextures(MAX_CACHED_TEXTURES / 4);
389
390 // Validate pixel data size
391 int expectedBytes = tileWidth * tileHeight * 4;
392 if (pixelData.Length < expectedBytes)
393 {
394 Console.WriteLine($"[SlideGLArea] UploadTileTexture: pixel data too small. Expected {expectedBytes} bytes for {tileWidth}x{tileHeight}, got {pixelData.Length}.");
395 return false;
396 }
397
398 // Create and upload texture
399 int tex = GL.GenTexture();
400 GL.BindTexture(TextureTarget.Texture2D, tex);
401 GL.PixelStore(PixelStoreParameter.UnpackAlignment, 1);
402
403 GL.TexImage2D(TextureTarget.Texture2D, 0, PixelInternalFormat.Rgba,
404 tileWidth, tileHeight, 0, PixelFormat.Rgba, PixelType.UnsignedByte, pixelData);
405
406 var glErr = GL.GetError();
407 if (glErr != ErrorCode.NoError)
408 {
409 Console.WriteLine($"[SlideGLArea] GL error after TexImage2D ({tileWidth}x{tileHeight}): {glErr}");
410 GL.DeleteTexture(tex);
411 return false;
412 }
413
414 GL.TexParameter(TextureTarget.Texture2D, TextureParameterName.TextureMinFilter, (int)TextureMinFilter.Linear);
415 GL.TexParameter(TextureTarget.Texture2D, TextureParameterName.TextureMagFilter, (int)TextureMagFilter.Linear);
416 GL.TexParameter(TextureTarget.Texture2D, TextureParameterName.TextureWrapS, (int)TextureWrapMode.ClampToEdge);
417 GL.TexParameter(TextureTarget.Texture2D, TextureParameterName.TextureWrapT, (int)TextureWrapMode.ClampToEdge);
418
419 GL.BindTexture(TextureTarget.Texture2D, 0);
420 _textureCache[index] = tex;
421 return true;
422 }
