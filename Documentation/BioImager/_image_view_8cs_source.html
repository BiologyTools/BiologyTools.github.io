551 private async void InitPreview()
552 {
553 if (SelectedImage == null)
554 return;
555 overviewPreviewBitmap?.Dispose();
556 overviewPreviewBitmap = null;
557 if (SelectedImage.Resolutions.Count == 0)
558 {
559 ShowOverview = false;
560 UpdateOverviewOverlay();
561 return;
562 }
563 int lev = GetPreviewLevel(SelectedImage);
564 int fullWidth = SelectedImage.Resolutions[0].SizeX;
565 int fullHeight = SelectedImage.Resolutions[0].SizeY;
566 double aspect = (double)fullWidth / Math.Max(1, fullHeight);
567 int overviewWidth = aspect >= 1.0
568 ? 120
569 : (int)Math.Max(1, Math.Round(120 * aspect));
570 int overviewHeight = aspect >= 1.0
571 ? (int)Math.Max(1, Math.Round(120 / aspect))
572 : 120;
573 overview = new Rectangle(0, 0, overviewWidth, overviewHeight);
574
575 ZCT coord = GetCoordinate();
576
577 byte[] bytes;
578 if (SelectedImage.Resolutions.Count == 1 && SelectedImage.Buffers.Count > 0)
579 {
580 using var sourceBitmap = ToSysBitmap(SelectedImage.Buffers[0]);
581 overviewPreviewBitmap = CreateScaledOverviewPreview(sourceBitmap, overviewWidth, overviewHeight);
582 ShowOverview = true;
583 UpdateOverviewOverlay();
584 return;
585 }
586 if (SelectedImage.Resolutions.Count == 1)
587 {
588 ShowOverview = false;
589 UpdateOverviewOverlay();
590 return;
591 }
592
593 var openSlideBase = SelectedImage.OpenSlideBase;
594 var slideBase = SelectedImage.SlideBase;
595 var sourceExtent = openSlideBase?.Schema?.Extent ?? slideBase?.Schema?.Extent;
596 if (!sourceExtent.HasValue || (openSlideBase == null && slideBase == null))
597 return;
598
599 if (openSlideBase != null)
600 {
601 int levelCount = openSlideBase.SlideImage.LevelCount;
602 if (levelCount <= 0)
603 return;
604
605 int sourceLevel = levelCount - 1;
606 var sourceDim = openSlideBase.SlideImage.GetLevelDimension(sourceLevel);
607 int sourceWidth = (int)sourceDim.Width;
608 int sourceHeight = (int)sourceDim.Height;
609 if (sourceWidth <= 0 || sourceHeight <= 0)
610 return;
611 var openSlideSliceInfo = new OpenSlideGTK.SliceInfo
612 {
613 Extent = sourceExtent.Value,
614 Resolution = Math.Max(sourceExtent.Value.Width / overviewWidth, sourceExtent.Value.Height / overviewHeight),
615 Parame = new OpenSlideGTK.SliceParame
616 {
617 DstPixelWidth = overviewWidth,
618 DstPixelHeight = overviewHeight,
619 },
620 };
621 bytes = await Task.Run(() => openSlideBase.GetSlice(openSlideSliceInfo, coord, sourceLevel));
622 if (bytes == null)
623 bytes = await Task.Run(() => openSlideBase.SlideImage.ReadRegion(sourceLevel, 0, 0, sourceWidth, sourceHeight));
624 if (bytes == null)
625 return;
626 using var sourceBitmap = bytes.Length == overviewWidth * overviewHeight * 4 ||
627 bytes.Length == overviewWidth * overviewHeight * 3 ||
628 bytes.Length == overviewWidth * overviewHeight * 2 ||
629 bytes.Length == overviewWidth * overviewHeight
630 ? CreateOverviewPreview(bytes, overviewWidth, overviewHeight)
631 : CreateOpenSlideOverviewPreview(bytes, sourceWidth, sourceHeight);
632 if (sourceBitmap == null)
633 return;
634 overviewSourceWidth = sourceWidth;
635 overviewSourceHeight = sourceHeight;
636 overviewSourceResolution = Math.Max(1e-9, SelectedImage.GetUnitPerPixel(sourceLevel));
637 overviewPreviewBitmap = CreateScaledOverviewPreview(sourceBitmap, overviewWidth, overviewHeight);
638 ShowOverview = true;
639 UpdateOverviewOverlay();
640 return;
641 }
642 else if (slideBase != null)
643 {
644 int sourceWidth = Math.Max(1, SelectedImage.Resolutions[lev].SizeX);
645 int sourceHeight = Math.Max(1, SelectedImage.Resolutions[lev].SizeY);
646 int frameIndex = SelectedImage.GetFrameIndex(coord.Z, coord.C, coord.T);
647 bytes = await Task.Run(() => SelectedImage.GetTileBytesRaw(
648 frameIndex,
649 lev,
650 0,
651 0,
652 sourceWidth,
653 sourceHeight,
654 coord).GetAwaiter().GetResult());
655 if (bytes != null && bytes.Length > 0)
656 {
657 using var sourceBitmap = CreateOverviewPreview(bytes, sourceWidth, sourceHeight);
658 if (sourceBitmap != null)
659 {
660 overviewSourceWidth = sourceWidth;
661 overviewSourceHeight = sourceHeight;
662 overviewSourceResolution = Math.Max(1e-9, SelectedImage.GetUnitPerPixel(lev));
663 overviewPreviewBitmap = CreateScaledOverviewPreview(sourceBitmap, overviewWidth, overviewHeight);
664 ShowOverview = true;
665 UpdateOverviewOverlay();
666 if (SelectedImage.isPyramidal)
667 UpdateView();
668 return;
669 }
670 }
671
672 var slideSliceInfo = new BioLib.SliceInfo
673 {
674 Extent = sourceExtent.Value,
675 Resolution = SelectedImage.GetUnitPerPixel(lev),
676 Coordinate = coord,
677 Parame = new BioLib.SliceParame
678 {
679 DstPixelWidth = overviewWidth,
680 DstPixelHeight = overviewHeight,
681 Quality = 100,
682 },
683 };
684 bytes = await Task.Run(() => slideBase.GetSlice(slideSliceInfo, new PointD(0, 0), new AForge.Size(overviewWidth, overviewHeight)));
685 }
686 else
687 return;
688 if (bytes == null)
689 return;
690 overviewSourceWidth = overviewWidth;
691 overviewSourceHeight = overviewHeight;
692 overviewSourceResolution = Math.Max(1e-9, SelectedImage.GetUnitPerPixel(lev));
693 overviewPreviewBitmap = CreateOverviewPreview(bytes, overviewWidth, overviewHeight);
694 if (overviewPreviewBitmap == null)
695 {
696 ShowOverview = false;
697 UpdateOverviewOverlay();
698 return;
699 }
700 ShowOverview = true;
701 UpdateOverviewOverlay();
702 if (SelectedImage.isPyramidal)
703 UpdateView();
704 }
