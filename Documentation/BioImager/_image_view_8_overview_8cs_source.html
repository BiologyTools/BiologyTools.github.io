106 private async void InitPreview()
107 {
108 BioImage image = OverviewSelectedImage;
109 if (image == null || !image.isPyramidal)
110 return;
111 var coordinate = (image.Coordinate.Z, image.Coordinate.C, image.Coordinate.T);
112 if (previewRequestImage == image && previewRequestCoordinate == coordinate)
113 return;
114 previewRequestImage = image;
115 previewRequestCoordinate = coordinate;
116 int version = ++previewVersion;
117 UpdateOverviewVisibility();
118 try
119 {
120 if (image.OpenSlideBase != null)
121 {
122 // OpenSlideBase.GetTile returns a schema-sized tile, not an arbitrary
123 // region. Choose a level that retains enough pixels for the overview.
124 var slide = image.OpenSlideBase;
125 var levels = slide.Schema.Resolutions.OrderBy(r => r.Value.UnitsPerPixel).ToArray();
126 var levelInfo = levels.Where(r =>
127 Math.Max(slide.Schema.Extent.Width, slide.Schema.Extent.Height) /
128 r.Value.UnitsPerPixel >= 160).DefaultIfEmpty(levels[0]).Last();
129 int width = (int)Math.Ceiling(slide.Schema.Extent.Width / levelInfo.Value.UnitsPerPixel);
130 int height = (int)Math.Ceiling(slide.Schema.Extent.Height / levelInfo.Value.UnitsPerPixel);
131 if ((long)width * height > 16_000_000)
132 return;
133 using Drawing.Bitmap slidePreview = await Task.Run(() => SlideOverviewBuilder.Build(width, height,
134 levelInfo.Value.TileWidth, levelInfo.Value.TileHeight,
135 (column, row) => slide.GetTile(new BruTile.TileInfo
136 {
137 Index = new BruTile.TileIndex(column, row, levelInfo.Key)
138 })));
139 if (version == previewVersion && !IsDisposed && !Disposing && image == OverviewSelectedImage)
140 SetOverviewPreview(slidePreview, image);
141 return;
142 }
143 // Use a native slide level, never the label or macro image.
144 int level = GetPreviewLevel(image);
145 var resolution = image.Resolutions[level];
146 if ((long)resolution.SizeX * resolution.SizeY > 16_000_000)
147 return;
148 int frame = image.GetFrameIndex(coordinate.Z, coordinate.C, coordinate.T);
149 using Bitmap tile = await image.GetTile(frame, level, 0, 0, resolution.SizeX, resolution.SizeY,
150 new ZCT(coordinate.Item1, coordinate.Item2, coordinate.Item3), true);
151 if (tile == null || version != previewVersion || IsDisposed || Disposing ||
152 image != OverviewSelectedImage)
153 return;
154 using Drawing.Bitmap display = ToSysBitmap(tile);
155 SetOverviewPreview(display, image);
156 }
157 catch (Exception ex)
158 {
159 System.Diagnostics.Trace.WriteLine($"Could not load slide overview: {ex.Message}");
160 }
161 }

180 private void OverviewPaint(object sender, PaintEventArgs e)
181 {
182 BioImage image = OverviewSelectedImage;
183 if (overviewImage == null || image == null || overviewSource != image)
184 return;
185 e.Graphics.DrawImageUnscaled(overviewImage, 0, 0);
186 RectangleD bounds = OverviewWorldBounds(image);
187 if (bounds.W <= 0 || bounds.H <= 0)
188 return;
189 // Use this viewer's transform even when a different tab is active.
190 PointD topLeft = image.isPyramidal
191 ? new PointD(image.PyramidalOrigin.X, image.PyramidalOrigin.Y)
192 : new PointD(-ViewWidth / 2 / (PxWmicron * Scale.Width) - Origin.X,
193 -ViewHeight / 2 / (PxHmicron * Scale.Height) - Origin.Y);
194 double viewWidth = image.isPyramidal ? ViewWidth * image.Resolution : ViewWidth / (PxWmicron * Scale.Width);
195 double viewHeight = image.isPyramidal ? ViewHeight * image.Resolution : ViewHeight / (PxHmicron * Scale.Height);
196 double left = Math.Clamp((topLeft.X - bounds.X) / bounds.W, 0, 1);
197 double top = Math.Clamp((topLeft.Y - bounds.Y) / bounds.H, 0, 1);
198 double right = Math.Clamp((topLeft.X + viewWidth - bounds.X) / bounds.W, 0, 1);
199 double bottom = Math.Clamp((topLeft.Y + viewHeight - bounds.Y) / bounds.H, 0, 1);
200 if (right > left && bottom > top)
201 {
202 using var pen = new Drawing.Pen(Drawing.Color.OrangeRed, 2);
203 e.Graphics.DrawRectangle(pen, (float)(left * (overview.Width - 1)),
204 (float)(top * (overview.Height - 1)), (float)((right - left) * (overview.Width - 1)),
205 (float)((bottom - top) * (overview.Height - 1)));
206 }
207 e.Graphics.DrawRectangle(Drawing.Pens.White, 0, 0, overview.Width - 1, overview.Height - 1);
208 }

213 if (image == null || overviewImage == null || overviewSource != image)
214 return;
215 App.viewer = this;
216 RectangleD bounds = OverviewWorldBounds(image);
217 double targetX = bounds.X + Math.Clamp(x / (double)overview.Width, 0, 1) * bounds.W;
218 double targetY = bounds.Y + Math.Clamp(y / (double)overview.Height, 0, 1) * bounds.H;
219 if (image.isPyramidal)
220 {
221 PyramidalOrigin = new PointD(targetX - ViewWidth * Resolution / 2.0,
222 targetY - ViewHeight * Resolution / 2.0);
223 UpdateImages();
224 }
225 else
