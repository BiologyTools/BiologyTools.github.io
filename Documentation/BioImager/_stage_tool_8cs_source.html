350 {

353 MessageBox.Show("'LiveView' GUI Property not set. Go to automation tab and create a property corresponding to the live view GUI element.");
354 return;

357 Microscope.SetFocus((double)upperLimBox.Value);
358 long max = long.MinValue;
359 double ud = (double)upperLimBox.Value;
360 for (double i = 0; i < (double)upperLimBox.Value; i += (double)focusInterval.Value)
361 {
362 Microscope.SetFocus((double)i);
363 System.Drawing.Graphics g;
364 System.Drawing.Bitmap b = new System.Drawing.Bitmap((int)re.Width, (int)re.Height, System.Drawing.Imaging.PixelFormat.Format32bppArgb);
365 g = System.Drawing.Graphics.FromImage(b);
366 g.CopyFromScreen(new System.Drawing.Point((int)re.X, (int)re.Y), new System.Drawing.Point(0, 0), new System.Drawing.Size(re.Width, re.Height));
367 Clipboard.SetImage(b);
368 System.Drawing.Imaging.BitmapData d = b.LockBits(new System.Drawing.Rectangle(0, 0, b.Width, b.Height), System.Drawing.Imaging.ImageLockMode.ReadWrite, System.Drawing.Imaging.PixelFormat.Format32bppArgb);
369 AForge.Bitmap bf = new AForge.Bitmap("", new UnmanagedImage(d.Scan0, b.Width, b.Height, b.Width * 4, PixelFormat.Format32bppArgb), new ZCT(), 0);
370 long dd = BioImage.CalculateFocusQuality(bf);
371 if (max < dd)
372 {
373 max = dd;
374 ud = (double)i;
