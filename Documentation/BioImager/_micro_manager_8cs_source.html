108 private static bool TakeImage(string file, out int width, out int height, out int depth, out int bits, out int channels, out AForge.Bitmap[] bm)
109 {
110 string fil = Path.GetFileNameWithoutExtension(file);
111 width = (int)core.getImageWidth();
112 height = (int)core.getImageHeight();
113 depth = (int)core.getImageBitDepth();
114 bits = (int)core.getBytesPerPixel();
115 channels = (int)core.getNumberOfCameraChannels();
116 PixelFormat format = PixelFormat.Format8bppIndexed;
117 if (depth > 8 && bits == 2)
118 format = PixelFormat.Format16bppGrayScale;
119 else if(depth > 8 && bits == 6)
120 format = PixelFormat.Format48bppRgb;
121 else if (depth == 8 && bits == 3)
122 format = PixelFormat.Format24bppRgb;
123 else if (depth == 8 && bits == 4)
124 format = PixelFormat.Format32bppArgb;
125 bm = new AForge.Bitmap[channels];
126 core.snapImage();
127 for (int i = 0; i < channels; i++)
128 {
129 if(depth <= 8)
130 {
131 byte[] bytes = (byte[])core.getImage(i);
132 bm[i] = new AForge.Bitmap(width, height, format, bytes , new ZCT(), "");
133 }
134 else
135 {
136 ushort[] buf = (ushort[])core.getImage(i);
137 byte[] bts = new byte[buf.Length * 2];
138 for (int b = 0; b < buf.Length; b++)
139 {
140 byte[] pl = BitConverter.GetBytes(buf[b]);
141 bts[b] = pl[0];
142 bts[b+1] = pl[1];
143 }
144 bm[i] = new AForge.Bitmap(width, height, format, bts, new ZCT(), "");
145 }
146 }
147 return true;
