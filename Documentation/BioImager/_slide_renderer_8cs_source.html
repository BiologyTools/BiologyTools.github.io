229 private async Task<byte[]> FetchTileAsync(TileInfo tileInfo, int level, ZCT coordinate,
230 ITileSchema schema, BioImage? rgbZarr, int[]? rgbSelection, TimeSpan timeout)
231 {
232 // Capture the source before awaiting: a tab can select another image while I/O is pending.
233 var openSource = _openSlideBase;
234 var slideSource = _slideBase;
235 try
236 {
237 using var cts = new CancellationTokenSource(timeout);
238 await _fetchSemaphore.WaitAsync(cts.Token);
239 try
240 {
241 byte[] data;
242 if (rgbZarr != null)
243 {
244 double units = schema.Resolutions[level].UnitsPerPixel;
245 int width = (int)Math.Round(tileInfo.Extent.Width / units);
246 int height = (int)Math.Round(tileInfo.Extent.Height / units);
247 data = await ZarrChannelView.ReadRgbTileAsync(rgbZarr, level,
248 tileInfo.Index.Col * schema.GetTileWidth(level),
249 tileInfo.Index.Row * schema.GetTileHeight(level),
250 width, height, coordinate, rgbSelection!).WaitAsync(cts.Token);
251 // SlideGLArea uploads BGRA; the direct Zarr reader returns RGBA.
252 for (int i = 0; i + 3 < data.Length; i += 4)
253 (data[i], data[i + 2]) = (data[i + 2], data[i]);
254 }
255 else
256 {
257 data = openSource != null
258 ? await openSource.GetTileAsync(tileInfo).WaitAsync(cts.Token)
259 : await slideSource.GetTileAsync(tileInfo, coordinate).WaitAsync(cts.Token);
260 if (data != null && rgbSelection != null &&
261 data.Length == schema.GetTileWidth(level) * schema.GetTileHeight(level) * 4)
262 {
263 data = (byte[])data.Clone();
264 for (int i = 0; i + 3 < data.Length; i += 4)
265 for (int color = 0; color < 3; color++)
266 if (rgbSelection[color] < 0) data[i + 2 - color] = 0;
267 }
268 }
269 return data;
270 }
271 finally { _fetchSemaphore.Release(); }
272 }
273 catch (Exception ex)
274 {
275 System.Diagnostics.Trace.WriteLine($"Could not fetch tile {tileInfo.Index}: {ex.Message}");
276 return null;
277 }

280 private async Task ProcessPendingFetchAsync(
281 TileInfo tile,
282 Task<byte[]> fetchTask,
283 PointD pyramidalOrigin,
284 double resolution,
285 int level,
286 ITileSchema schema,
287 int contentVersion,
288 bool directZarr)
289 {
290 try
291 {
292 byte[] tileData = await fetchTask.ConfigureAwait(false);
293 if (tileData == null || contentVersion != _contentVersion)
294 return;
295
296 double units = schema.Resolutions[level].UnitsPerPixel;
297 int tW = (int)Math.Round(tile.Extent.Width / units);
298 int tH = (int)Math.Round(tile.Extent.Height / units);
299 if (tW <= 0 || tH <= 0) return;
300 if (!directZarr)
301 {
302 int sourceWidth = schema.GetTileWidth(level);
303 int sourceHeight = schema.GetTileHeight(level);
304 // Edge TileInfo extents can be smaller than the padded source buffer.
305 int pixels = sourceWidth * sourceHeight;
306 int bytesPerPixel = pixels > 0 && tileData.Length % pixels == 0 ? tileData.Length / pixels : 0;
307 if ((tW != sourceWidth || tH != sourceHeight) && bytesPerPixel is >= 1 and <= 4)
308 {
309 byte[] clipped = new byte[checked(tW * tH * bytesPerPixel)];
310 int rowBytes = tW * bytesPerPixel;
311 for (int row = 0; row < tH; row++)
312 Buffer.BlockCopy(tileData, row * sourceWidth * bytesPerPixel, clipped, row * rowBytes, rowBytes);
313 tileData = clipped;
314 }
315 }
316
317 await RunOnUiThreadAsync(() =>
318 {
319 if (contentVersion != _contentVersion) return;
320 if (_glArea.UploadTileTexture(tile.Index, tileData, tW, tH))
321 {
322 _uploadedTiles.Add(tile.Index);
323 _glArea.RequestRedraw();
324 }
325 });
326 }
327 catch
328 {
329 }
330 finally
331 {
332 lock (_pendingTileFetchLock)
333 {
334 _pendingTileFetches.Remove((tile.Index, contentVersion));
335 }
336
337 if (Interlocked.Exchange(ref _redrawQueued, 1) == 0)
338 {
339 try
340 {
341 await RunOnUiThreadAsync(() => _glArea.RequestRedraw());
342 }
343 finally
344 {
345 Interlocked.Exchange(ref _redrawQueued, 0);
346 }
347 }
348 }
349 }

386 private TileRenderInfo CalculateScreenPosition(
387 TileInfo tile,
388 PointD pyramidalOrigin,
389 double viewResolution,
390 int level,
391 ITileSchema schema)
392 {
393 var tileExtent = tile.Extent;
394 var clippedExtent = new Extent(
395 Math.Max(tileExtent.MinX, schema.Extent.MinX),
396 Math.Max(tileExtent.MinY, schema.Extent.MinY),
397 Math.Min(tileExtent.MaxX, schema.Extent.MaxX),
398 Math.Min(tileExtent.MaxY, schema.Extent.MaxY));
399
400 if (clippedExtent.MaxX <= clippedExtent.MinX || clippedExtent.MaxY <= clippedExtent.MinY)
401 return new TileRenderInfo(tile.Index, -9999, -9999, 0, 0);
402
403 double originScreenX = pyramidalOrigin.X / viewResolution;
404 double originScreenY = pyramidalOrigin.Y / viewResolution;
405
406 float screenX = (float)(clippedExtent.MinX / viewResolution - originScreenX);
407 float screenY = (float)(-clippedExtent.MaxY / viewResolution - originScreenY);
408 float screenWidth = (float)((clippedExtent.MaxX - clippedExtent.MinX) / viewResolution);
409 float screenHeight = (float)((clippedExtent.MaxY - clippedExtent.MinY) / viewResolution);
410
411 return new TileRenderInfo(tile.Index, screenX, screenY, screenWidth, screenHeight);
412 }
