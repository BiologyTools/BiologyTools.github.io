173 public bool HasTab(BioImage image)
174 {
175 if (image == null)
176 return false;
177 foreach (TabPage tabPage in tabControl.TabPages)
178 {
179 if (tabPage.Controls.Count == 0 || tabPage.Controls[0] is not ImageView view)
180 continue;
181 foreach (BioImage tabImage in view.Images)
182 {
183 if (tabImage == null)
184 continue;
185 if (ReferenceEquals(tabImage, image))
186 return true;
187 if (!string.IsNullOrEmpty(tabImage.ID) && !string.IsNullOrEmpty(image.ID) && tabImage.ID == image.ID)
188 return true;
189 if (!string.IsNullOrEmpty(tabImage.Filename) && !string.IsNullOrEmpty(image.Filename) && tabImage.Filename == image.Filename)
190 return true;
191 }
192 }
193 return false;
194 }

210 private static BioImage CreateAssociatedImageTab(BioImage source, string name, AssociatedImage image)
211 {
212 if (source == null || image == null)
213 return null;
214
215 string baseName = Path.GetFileNameWithoutExtension(source.Filename);
216 if (string.IsNullOrEmpty(baseName))
217 baseName = Path.GetFileNameWithoutExtension(source.ID);
218 if (string.IsNullOrEmpty(baseName))
219 baseName = "associated";
220
221 string suffix = name.Trim();
222 if (string.IsNullOrEmpty(suffix))
223 suffix = "associated";
224
225 BioImage bi = new BioImage($"{source.ID}::{suffix}");
226 bi.Filename = $"{baseName}-{suffix}.png";
227 bi.file = source.file;
228 bi.Resolutions.Add(new Resolution(
229 (int)image.Dimensions.Width,
230 (int)image.Dimensions.Height,
231 AForge.PixelFormat.Format32bppArgb,
232 1,
233 1,
234 1,
235 0,
236 0,
237 0));
238 bi.Buffers.Add(new AForge.Bitmap(
239 "",
240 (int)image.Dimensions.Width,
241 (int)image.Dimensions.Height,
242 AForge.PixelFormat.Format32bppArgb,
243 image.Data,
244 new ZCT(),
245 0,
246 null,
247 littleEndian: true,
248 interleaved: true));
249 bi.Channels.Add(new Channel(0, 8, 4));
250 bi.UpdateCoords(1, 1, 1);
251 bi.Volume = new VolumeD(new Point3D(0, 0, 0), new Point3D(
252 bi.PhysicalSizeX * bi.SizeX,
253 bi.PhysicalSizeY * bi.SizeY,
254 bi.PhysicalSizeZ * bi.SizeZ));
255 bi.Coordinate = new ZCT(0, 0, 0);
256 return bi;
257 }
258 private void AddAssociatedImageTabs(BioImage source)
259 {
260 if (source?.OpenSlideBase?.SlideImage == null)
261 return;
262
263 foreach (var entry in source.OpenSlideBase.SlideImage.GetAssociatedImages())
264 {
265 string name = entry.Key ?? string.Empty;
266 if (!name.Contains("label", StringComparison.OrdinalIgnoreCase) &&
267 !name.Contains("macro", StringComparison.OrdinalIgnoreCase))
268 {
269 continue;
270 }
271
272 BioImage associated = CreateAssociatedImageTab(source, name, entry.Value);
273 if (associated != null && !HasTab(associated))
274 AddTabCore(associated);
275 }

466 private static void ProgressUpdate()
467 {
468 do
469 {
470 progress.Text = title;

473 Thread.Sleep(250);
474 } while (!done);

476 private static void StartProgress(string titl, string status)
477 {
478 done = false;
479 title = titl;
480 BioImage.Progress = 0;
481 progress.Status = status;
482 progress.Text = title;
483 if (BioImage.Status == "" || BioImage.Status == null)
484 progress.Status = status;
485 else
486 progress.Status = BioImage.Status;
487 progress.ProgressValue = BioImage.Progress;
488 progress.Show();
489 Application.DoEvents();
490 Thread th = new Thread(ProgressUpdate);
491 th.Start();
492 }
493 private static void StopProgress()
494 {
495 done = true;
496 progress.Hide();

1143 if (e.KeyCode == Keys.S || e.KeyCode == Keys.NumPad2)
1144 {
1145 Viewer.Origin = new PointD(Viewer.Origin.X, Viewer.Origin.Y + moveAmount);
1146 }
1147 if (e.KeyCode == Keys.A || e.KeyCode == Keys.NumPad4)
1148 {
1149 Viewer.Origin = new PointD(Viewer.Origin.X - moveAmount, Viewer.Origin.Y);
1150 }
1151 if (e.KeyCode == Keys.D || e.KeyCode == Keys.NumPad6)
1152 {
1153 Viewer.Origin = new PointD(Viewer.Origin.X + moveAmount, Viewer.Origin.Y);
1154 }
1155 if (e.KeyCode == Keys.Up)
1156 Microscope.MoveUp(Microscope.viewSize.Y / 2);
1157 if (e.KeyCode == Keys.Down)
1158 Microscope.MoveDown(Microscope.viewSize.Y / 2);
1159 if (e.KeyCode == Keys.Left)
