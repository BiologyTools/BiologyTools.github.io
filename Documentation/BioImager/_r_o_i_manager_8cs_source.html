171 private void roiView_SelectedIndexChanged(object sender, EventArgs e)

175 ListViewItem it = roiView.SelectedItems[0];
176 anno = (ROI)it.Tag;
177 if(App.viewer!=null)
178 App.viewer.SetCoordinate(anno.coord.Z, anno.coord.C, anno.coord.T);
179 if(anno.type == ROI.Type.Line || anno.type == ROI.Type.Polygon ||
180 anno.type == ROI.Type.Polyline)
181 {
182 xBox.Enabled = false;
183 yBox.Enabled = false;
184 wBox.Enabled = false;
185 hBox.Enabled = false;
186 }
187 else
188 {
189 xBox.Enabled = true;
190 yBox.Enabled = true;
191 wBox.Enabled = true;
192 hBox.Enabled = true;
193 }
194 if(anno.type == ROI.Type.Rectangle || anno.type == ROI.Type.Ellipse)
195 {
196 pointIndexBox.Enabled = false;
197 pointXBox.Enabled = false;
198 pointYBox.Enabled = false;
199 }
200 else
201 {
202 pointIndexBox.Enabled = true;
203 pointXBox.Enabled = true;
204 pointYBox.Enabled = true;
205 }
206 xBox.Value = (decimal)anno.X;
207 yBox.Value = (decimal)anno.Y;
208 wBox.Value = (decimal)anno.W;
209 hBox.Value = (decimal)anno.H;
210 zBox.Value = anno.coord.Z;
211 cBox.Value = anno.coord.C;
212 tBox.Value = anno.coord.T;
213 rBox.Value = anno.strokeColor.R;
214 gBox.Value = anno.strokeColor.G;
215 bBox.Value = anno.strokeColor.B;
216 strokeWBox.Value = (decimal)anno.strokeWidth;
217 idBox.Text = anno.id;
218 textBox.Text = anno.Text;
219 typeBox.SelectedIndex = (int)anno.type;
220 UpdatePointBox();
