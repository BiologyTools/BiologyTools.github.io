83 private const string VertexShaderSource = @"
84#version 330 core
85
86layout(location=0) in vec2 aPos;
87layout(location=1) in vec2 aUV;
88
89uniform vec2 pos; // pixel-space top-left of tile
90uniform vec2 size; // pixel-space size of tile
91uniform vec2 viewportSize; // (width, height) in pixels
92uniform vec2 uvMin; // UV sub-region min (for boundary clipping)
93uniform vec2 uvMax; // UV sub-region max (for boundary clipping)
94
95out vec2 uv;
96
97void main()
98{
99 // aPos is in [0,1] x [0,1]
100 vec2 pixelPos = aPos * size + pos;
101
102 // Convert from pixel coords to NDC (-1..1), with Y flipped for GL
103 vec2 ndc;
104 ndc.x = (pixelPos.x / viewportSize.x) * 2.0 - 1.0;
105 ndc.y = 1.0 - (pixelPos.y / viewportSize.y) * 2.0;
106
107 gl_Position = vec4(ndc, 0.0, 1.0);
108
109 // Map aUV [0,1] to the sub-region [uvMin, uvMax]
110 uv = uvMin + aUV * (uvMax - uvMin);
111}
112";
113
114 private const string FragmentShaderSource = @"
115#version 330 core
116
117in vec2 uv;
118out vec4 FragColor;
119uniform sampler2D tex;
120
121void main()
122{
123 // Fix: Removed '1.0 - uv.y'.
124 // Image data is usually Top-Down. OpenGL Textures are Bottom-Up.
125 // This implicit flip means UV(0,0) (Top-Left Quad) maps to Texture Bottom (Image Top).
126 // This is the correct orientation for Top-Left origin rendering.
127 FragColor = texture(tex, vec2(uv.x, uv.y));
128}
129";

185 private void InitializeGL()
186 {
187 // Compile shaders
188 int vertexShader = CompileShader(ShaderType.VertexShader, VertexShaderSource);
189 int fragmentShader = CompileShader(ShaderType.FragmentShader, FragmentShaderSource);
190
191 _shaderProgram = GL.CreateProgram();
192 GL.AttachShader(_shaderProgram, vertexShader);
193 GL.AttachShader(_shaderProgram, fragmentShader);
194 GL.LinkProgram(_shaderProgram);
195
196 GL.GetProgram(_shaderProgram, GetProgramParameterName.LinkStatus, out int linkStatus);
197 if (linkStatus == 0)
198 {
199 string infoLog = GL.GetProgramInfoLog(_shaderProgram);
200 throw new Exception($"Shader link failed: {infoLog}");
201 }
202
203 GL.DeleteShader(vertexShader);
204 GL.DeleteShader(fragmentShader);
205
206 // Cache uniform locations
207 _locPos = GL.GetUniformLocation(_shaderProgram, "pos");
208 _locSize = GL.GetUniformLocation(_shaderProgram, "size");
209 _locViewportSize = GL.GetUniformLocation(_shaderProgram, "viewportSize");
210 _locTex = GL.GetUniformLocation(_shaderProgram, "tex");
211 _locUvMin = GL.GetUniformLocation(_shaderProgram, "uvMin");
212 _locUvMax = GL.GetUniformLocation(_shaderProgram, "uvMax");
213
214 // Create quad VAO/VBO for tile rendering
215 float[] quadVertices =
216 {
217 // pos uv
218 0, 0, 0, 0,
219 1, 0, 1, 0,
220 1, 1, 1, 1,
221
222 0, 0, 0, 0,
223 1, 1, 1, 1,
224 0, 1, 0, 1
225 };
226
227 _vao = GL.GenVertexArray();
228 _vbo = GL.GenBuffer();
229
230 GL.BindVertexArray(_vao);
231 GL.BindBuffer(BufferTarget.ArrayBuffer, _vbo);
232 GL.BufferData(BufferTarget.ArrayBuffer, quadVertices.Length * sizeof(float),
233 quadVertices, BufferUsageHint.StaticDraw);
234
235 // Position attribute
236 GL.VertexAttribPointer(0, 2, VertexAttribPointerType.Float, false, 4 * sizeof(float), 0);
237 GL.EnableVertexAttribArray(0);
238
239 // UV attribute
240 GL.VertexAttribPointer(1, 2, VertexAttribPointerType.Float, false, 4 * sizeof(float), 2 * sizeof(float));
241 GL.EnableVertexAttribArray(1);
242
243 GL.BindVertexArray(0);
244 }

352 private void RenderTiles(int width, int height)
353 {
354 int scale = ScaleFactor;
355 GL.Viewport(0, 0, width * scale, height * scale);
356 if (TilesToRender.Count == 0)
357 {
358 if (!PreservePreviousFrameWhileIncomplete)
359 {
360 GL.ClearColor(0.2f, 0.2f, 0.2f, 1f);
361 GL.Clear(ClearBufferMask.ColorBufferBit);
362 }
363 LogDiag($"[RenderTiles] no tiles to render viewport={width}x{height} imageScreen=({ImageScreenX},{ImageScreenY},{ImageScreenW},{ImageScreenH})");
364 return;
365 }
366
367 bool hasAllRenderableTextures = TilesToRender.All(tile => _textureCache.ContainsKey(tile.Index));
368 if (!hasAllRenderableTextures && PreservePreviousFrameWhileIncomplete)
369 {
370 // Keep the previous frame visible while the current visible set
371 // is still incomplete. Clearing here exposes the background for a
372 // frame whenever a view change lands before every tile is ready.
373 LogDiag($"[RenderTiles] waiting for complete tile set viewport={width}x{height} tileCount={TilesToRender.Count}");
374 return;
375 }
376
377 if (!hasAllRenderableTextures)
378 {
379 GL.ClearColor(0.2f, 0.2f, 0.2f, 1f);
380 GL.Clear(ClearBufferMask.ColorBufferBit);
381 }
382
383 GL.ClearColor(0.2f, 0.2f, 0.2f, 1f);
384 GL.Clear(ClearBufferMask.ColorBufferBit);
385
386 LogDiag($"[RenderTiles] tiles={TilesToRender.Count} viewport={width}x{height} imageScreen=({ImageScreenX},{ImageScreenY},{ImageScreenW},{ImageScreenH})");
387
388 GL.Enable(EnableCap.Blend);
389 GL.BlendFunc(BlendingFactor.SrcAlpha, BlendingFactor.OneMinusSrcAlpha);
390
391 // Scissor to the image boundary so tiles cannot render pixels outside the image.
392 // GL scissor Y is from bottom; convert from top-left screen coords.
393 if (ImageScreenW > 0 && ImageScreenH > 0)
394 {
395 GL.Enable(EnableCap.ScissorTest);
396 // Apply (int) cast AFTER multiplying by the HiDPI scale factor so
397 // sub-pixel rounding errors don't compound.
398 int sx = (int)(Math.Max(0, ImageScreenX) * scale);
399 int sy = (int)(Math.Max(0, (height - ImageScreenY - ImageScreenH)) * scale);
400 int sw = (int)(Math.Min(ImageScreenW, width - ImageScreenX) * scale);
401 int sh = (int)(Math.Min(ImageScreenH, height - ImageScreenY) * scale);
402 GL.Scissor(sx, sy, Math.Max(0, sw), Math.Max(0, sh));
403 }
404
405 GL.UseProgram(_shaderProgram);
406 GL.BindVertexArray(_vao);
407
408 GL.Uniform2(_locViewportSize, (float)width, (float)height);
409 GL.Uniform1(_locTex, 0);
410 GL.ActiveTexture(TextureUnit.Texture0);
411
412 int renderedCount = 0;
413 foreach (var tile in TilesToRender)
414 {
415 if (!_textureCache.TryGetValue(tile.Index, out int texId))
416 {
417 LogDiag($"[RenderTiles] missing texture tile={tile.Index}");
418 continue;
419 }
420
421 GL.Uniform2(_locPos, tile.ScreenX, tile.ScreenY);
422 GL.Uniform2(_locSize, tile.ScreenWidth, tile.ScreenHeight);
423 GL.Uniform2(_locUvMin, tile.U0, tile.V0);
424 GL.Uniform2(_locUvMax, tile.U1, tile.V1);
425
426 LogDiag($"[RenderTiles] draw tile={tile.Index} pos=({tile.ScreenX:F1},{tile.ScreenY:F1}) size=({tile.ScreenWidth:F1},{tile.ScreenHeight:F1}) tex={texId}");
427
428 GL.BindTexture(TextureTarget.Texture2D, texId);
429 GL.DrawArrays(PrimitiveType.Triangles, 0, 6);
430 renderedCount++;
431 }
432
433 LogDiag($"[RenderTiles] rendered={renderedCount}");
434
435 GL.Disable(EnableCap.ScissorTest);
436 GL.BindVertexArray(0);
437 GL.UseProgram(0);
438 GL.Disable(EnableCap.Blend);
439 }

458 public void UploadTileTexture(TileIndex index, byte[] pixelData, int tileWidth, int tileHeight)
459 {
460 if (!_glInitialized)
461 {
462 LogDiag($"[UploadTileTexture] gl-not-initialized tile={index}");
463 return;
464 }
465
466 if (pixelData == null || pixelData.Length == 0)
467 {
468 LogDiag($"[UploadTileTexture] empty pixel data tile={index}");
469 return;
470 }
471
472 // The buffer is always BGRA = 4 bytes per pixel.
473 // GetTile pads every tile — including edge tiles — to exactly
474 // tileWidth x tileHeight pixels, so the dimensions passed by SlideRenderer
475 // (the schema tile size) always match the buffer. We just validate and bail
476 // on any genuine mismatch rather than trying to guess new dimensions.
477 if (tileWidth <= 0 || tileHeight <= 0)
478 {
479 LogDiag($"[UploadTileTexture] invalid dimensions tile={index} size={tileWidth}x{tileHeight}");
480 return;
481 }
482
483 int expectedBytes = tileWidth * tileHeight * 4;
484 if (pixelData.Length < expectedBytes)
485 {
486 LogDiag($"[UploadTileTexture] buffer too small tile={index} need={expectedBytes} got={pixelData.Length}");
487 return;
488 }
489
490 MakeCurrent();
491
492 // Check if already cached
493 if (_textureCache.ContainsKey(index))
494 {
495 LogDiag($"[UploadTileTexture] already cached tile={index}");
496 return;
497 }
498
499 long newTextureBytes = (long)tileWidth * tileHeight * 4;
500
501 // Evict old textures if the cache is full or if the new upload would
502 // push the total texture budget over its byte cap.
503 while (_textureCache.Count > 0 &&
504 (_textureCache.Count >= MAX_CACHED_TEXTURES ||
505 _textureBytes + newTextureBytes > MAX_CACHED_TEXTURE_BYTES))
506 {
507 EvictOldestTextures(1);
508 }
509
510 // Create and upload texture.
511 // Pin the managed array explicitly so the GC cannot relocate it while the
512 // driver is reading from the pointer (avoids ExecutionEngineException).
513 // PixelFormat.Bgra matches the byte order produced by ReadRegionAsync.
514 int tex = GL.GenTexture();
515 GL.BindTexture(TextureTarget.Texture2D, tex);
516 GL.PixelStore(PixelStoreParameter.UnpackAlignment, 4);
517
518 var handle = System.Runtime.InteropServices.GCHandle.Alloc(pixelData, System.Runtime.InteropServices.GCHandleType.Pinned);
519 try
520 {
521 GL.TexImage2D(TextureTarget.Texture2D, 0, PixelInternalFormat.Rgba,
522 tileWidth, tileHeight, 0,
523 PixelFormat.Bgra, PixelType.UnsignedByte,
524 handle.AddrOfPinnedObject());
525 }
526 finally
527 {
528 handle.Free();
529 }
530
531 GL.TexParameter(TextureTarget.Texture2D, TextureParameterName.TextureMinFilter, (int)TextureMinFilter.Linear);
532 GL.TexParameter(TextureTarget.Texture2D, TextureParameterName.TextureMagFilter, (int)TextureMagFilter.Linear);
533 GL.TexParameter(TextureTarget.Texture2D, TextureParameterName.TextureWrapS, (int)TextureWrapMode.ClampToEdge);
534 GL.TexParameter(TextureTarget.Texture2D, TextureParameterName.TextureWrapT, (int)TextureWrapMode.ClampToEdge);
535
536 GL.BindTexture(TextureTarget.Texture2D, 0);
537
538 _textureCache[index] = tex;
539 _textureSizes[index] = (tileWidth, tileHeight);
540 _textureBytes += newTextureBytes;
541 LogDiag($"[UploadTileTexture] uploaded tile={index} tex={tex} size={tileWidth}x{tileHeight} bytes={pixelData.Length}");
542 }
