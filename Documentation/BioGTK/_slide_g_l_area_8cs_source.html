89 private const string VertexShaderSource = @"
90#version 330 core
91
92layout(location=0) in vec2 aPos;
93layout(location=1) in vec2 aUV;
94
95uniform vec2 pos; // pixel-space top-left of tile
96uniform vec2 size; // pixel-space size of tile
97uniform vec2 viewportSize; // (width, height) in pixels
98uniform vec2 uvMin; // UV sub-region min (for boundary clipping)
99uniform vec2 uvMax; // UV sub-region max (for boundary clipping)
100
101out vec2 uv;
102
103void main()
104{
105 // aPos is in [0,1] x [0,1]
106 vec2 pixelPos = aPos * size + pos;
107
108 // Convert from pixel coords to NDC (-1..1), with Y flipped for GL
109 vec2 ndc;
110 ndc.x = (pixelPos.x / viewportSize.x) * 2.0 - 1.0;
111 ndc.y = 1.0 - (pixelPos.y / viewportSize.y) * 2.0;
112
113 gl_Position = vec4(ndc, 0.0, 1.0);
114
115 // Map aUV [0,1] to the sub-region [uvMin, uvMax]
116 uv = uvMin + aUV * (uvMax - uvMin);
117}
118";
119
120 private const string FragmentShaderSource = @"
121#version 330 core
122
123in vec2 uv;
124out vec4 FragColor;
125uniform sampler2D tex;
126
127void main()
128{
129 // Fix: Removed '1.0 - uv.y'.
130 // Image data is usually Top-Down. OpenGL Textures are Bottom-Up.
131 // This implicit flip means UV(0,0) (Top-Left Quad) maps to Texture Bottom (Image Top).
132 // This is the correct orientation for Top-Left origin rendering.
133 FragColor = texture(tex, vec2(uv.x, uv.y));
134}
135";

196 private void InitializeGL()
197 {
198 // Compile shaders
199 int vertexShader = CompileShader(ShaderType.VertexShader, VertexShaderSource);
200 int fragmentShader = CompileShader(ShaderType.FragmentShader, FragmentShaderSource);
201
202 _shaderProgram = GL.CreateProgram();
203 GL.AttachShader(_shaderProgram, vertexShader);
204 GL.AttachShader(_shaderProgram, fragmentShader);
205 GL.LinkProgram(_shaderProgram);
206
207 GL.GetProgram(_shaderProgram, GetProgramParameterName.LinkStatus, out int linkStatus);
208 if (linkStatus == 0)
209 {
210 string infoLog = GL.GetProgramInfoLog(_shaderProgram);
211 throw new Exception($"Shader link failed: {infoLog}");
212 }
213
214 GL.DeleteShader(vertexShader);
215 GL.DeleteShader(fragmentShader);
216
217 // Cache uniform locations
218 _locPos = GL.GetUniformLocation(_shaderProgram, "pos");
219 _locSize = GL.GetUniformLocation(_shaderProgram, "size");
220 _locViewportSize = GL.GetUniformLocation(_shaderProgram, "viewportSize");
221 _locTex = GL.GetUniformLocation(_shaderProgram, "tex");
222 _locUvMin = GL.GetUniformLocation(_shaderProgram, "uvMin");
223 _locUvMax = GL.GetUniformLocation(_shaderProgram, "uvMax");
224
225 // Create quad VAO/VBO for tile rendering
226 float[] quadVertices =
227 {
228 // pos uv
229 0, 0, 0, 0,
230 1, 0, 1, 0,
231 1, 1, 1, 1,
232
233 0, 0, 0, 0,
234 1, 1, 1, 1,
235 0, 1, 0, 1
236 };
237
238 _vao = GL.GenVertexArray();
239 _vbo = GL.GenBuffer();
240
241 GL.BindVertexArray(_vao);
242 GL.BindBuffer(BufferTarget.ArrayBuffer, _vbo);
243 GL.BufferData(BufferTarget.ArrayBuffer, quadVertices.Length * sizeof(float),
244 quadVertices, BufferUsageHint.StaticDraw);
245
246 // Position attribute
247 GL.VertexAttribPointer(0, 2, VertexAttribPointerType.Float, false, 4 * sizeof(float), 0);
248 GL.EnableVertexAttribArray(0);
249
250 // UV attribute
251 GL.VertexAttribPointer(1, 2, VertexAttribPointerType.Float, false, 4 * sizeof(float), 2 * sizeof(float));
252 GL.EnableVertexAttribArray(1);
253
254 GL.BindVertexArray(0);
255 }

258 {
259 int shader = GL.CreateShader(type);
260 GL.ShaderSource(shader, source);
261 GL.CompileShader(shader);
262
263 GL.GetShader(shader, ShaderParameter.CompileStatus, out int status);
264 if (status == 0)
265 {
266 string infoLog = GL.GetShaderInfoLog(shader);
267 throw new Exception($"Shader compilation failed ({type}): {infoLog}");
268 }
269
270 return shader;
271 }

301 private void InitializeSkia(int framebuffer = -1)
302 {
303 // Capture GTK's framebuffer before disposing the previous Skia target;
304 // target disposal is allowed to change GL bindings.
305 if (framebuffer < 0)
306 GL.GetInteger(GetPName.FramebufferBinding, out framebuffer);
307
308 // Keep one GPU context for the lifetime of the GLArea. Creating a new
309 // GRContext on every frame leaks native GL resources and eventually
310 // causes allocations such as GL.GenTexture to return 0.
311 _grContext ??= GRContext.CreateGl();
312
313 // GTK may rotate the GLArea framebuffer between renders. Recreate only
314 // the lightweight target/surface wrappers for the currently bound FBO.
315 _skSurface?.Dispose();
316 _skSurface = null;
317 _renderTarget?.Dispose();
318 _renderTarget = null;
319
320 int targetWidth = AllocatedWidth * ScaleFactor;
321 int targetHeight = AllocatedHeight * ScaleFactor;
322 if (_grContext == null || targetWidth <= 0 || targetHeight <= 0)
323 return;
324
325 // Ensure the format matches what Gtk.GLArea provides (usually GL_RGBA8)
326 var framebufferInfo = new GRGlFramebufferInfo((uint)framebuffer, 0x8058); // GL_RGBA8
327
328 _renderTarget = new GRBackendRenderTarget(
329 targetWidth,
330 targetHeight,
331 0, 8, framebufferInfo);
332
333 _skSurface = SKSurface.Create(_grContext, _renderTarget, GRSurfaceOrigin.BottomLeft, SKColorType.Rgba8888);
334 }
335
336 private void CleanupSkia()
337 {
338 _skSurface?.Dispose();
339 _skSurface = null;
340
341 _renderTarget?.Dispose();
342 _renderTarget = null;
343
344 _grContext?.Dispose();
345 _grContext = null;
346 }

352 private void OnRender(object sender, RenderArgs args)
353 {
354 if (!_glInitialized)
355 {
356 args.RetVal = true;
357 return;
358 }
359
360 MakeCurrent();
361 GL.GetInteger(GetPName.FramebufferBinding, out int presentationFramebuffer);
362
363 int width = AllocatedWidth;
364 int height = AllocatedHeight;
365
366 if (width <= 0 || height <= 0)
367 {
368 args.RetVal = true;
369 return;
370 }
371
372 // Phase 1: Render tiles with OpenGL
373 RenderTiles(width, height);
374 if (_tileFrameComplete)
375 CaptureCompleteTileFrame(width, height);
376
377 // The initial fit-to-window frame already contains everything needed
378 // by the minimap. Capture it before Skia adds the overview and ROIs so
379 // ImageView does not perform a second remote pyramid read.
380 CaptureWholeImageFrameIfRequested(width, height);
381
382 // Phase 2: Render annotations with Skia
383 try
384 {
385 RenderSkiaOverlay(width, height, presentationFramebuffer);
386 }
387 catch (Exception ex)
388 {
389 Console.WriteLine($"Skia overlay render error: {ex.Message}");
390 }
391
392 // Skia surfaces are alpha-capable even though the viewer window is not.
393 // Some Windows GL/GDK combinations leave untouched overlay pixels with
394 // alpha zero. Rewrite only the alpha channel after the overlay so the
395 // compositor can never expose windows behind the viewer.
396 GL.BindFramebuffer(FramebufferTarget.Framebuffer, presentationFramebuffer);
397 ForceOpaqueFramebuffer();
398 GL.Flush();
399 args.RetVal = true;
400 }

402 private static void ForceOpaqueFramebuffer()
403 {
404 GL.Disable(EnableCap.ScissorTest);
405 GL.ColorMask(false, false, false, true);
406 GL.ClearColor(0f, 0f, 0f, 1f);
407 GL.Clear(ClearBufferMask.ColorBufferBit);
408 GL.ColorMask(true, true, true, true);
409 }

411 private void RenderTiles(int width, int height)
412 {
413 _tileFrameComplete = false;
414 int scale = ScaleFactor;
415 GL.Viewport(0, 0, width * scale, height * scale);
416
417 // GLArea may provide a different framebuffer for each render. Its
418 // previous contents are undefined, so returning early in an attempt
419 // to preserve the prior frame can present transparent pixels to the
420 // window compositor. Establish a fully opaque frame on every pass;
421 // cached fallback tiles are drawn over it below when available.
422 GL.Disable(EnableCap.ScissorTest);
423 GL.ColorMask(true, true, true, true);
424 GL.ClearColor(0.2f, 0.2f, 0.2f, 1f);
425 GL.Clear(ClearBufferMask.ColorBufferBit);
426
427 if (TilesToRender.Count == 0)
428 {
429 if (PreservePreviousFrameWhileIncomplete)
430 RenderCompleteFrameFallback(width, height);
431 LogDiag($"[RenderTiles] no tiles to render viewport={width}x{height} imageScreen=({ImageScreenX},{ImageScreenY},{ImageScreenW},{ImageScreenH})");
432 return;
433 }
434
435 bool hasAllRenderableTextures = TilesToRender.All(tile => _textureCache.ContainsKey(tile.Index));
436 if (!hasAllRenderableTextures)
437 {
438 // Never expose a partially assembled target frame. Retain the last
439 // complete GPU snapshot until all visible tiles are resident, then
440 // replace the frame atomically. On first load, keep the opaque
441 // background instead of popping tiles in one at a time.
442 if (_completeFrameTexture != 0)
443 RenderCompleteFrameFallback(width, height);
444 LogDiag($"[RenderTiles] waiting for atomic tile frame viewport={width}x{height} tileCount={TilesToRender.Count}");
445 return;
446 }
447
448 LogDiag($"[RenderTiles] tiles={TilesToRender.Count} viewport={width}x{height} imageScreen=({ImageScreenX},{ImageScreenY},{ImageScreenW},{ImageScreenH})");
449
450 GL.Enable(EnableCap.Blend);
451 GL.BlendFunc(BlendingFactor.SrcAlpha, BlendingFactor.OneMinusSrcAlpha);
452
453 // Scissor to the image boundary so tiles cannot render pixels outside the image.
454 // GL scissor Y is from bottom; convert from top-left screen coords.
455 if (ImageScreenW > 0 && ImageScreenH > 0)
456 {
457 GL.Enable(EnableCap.ScissorTest);
458 // Apply (int) cast AFTER multiplying by the HiDPI scale factor so
459 // sub-pixel rounding errors don't compound.
460 int sx = (int)(Math.Max(0, ImageScreenX) * scale);
461 int sy = (int)(Math.Max(0, (height - ImageScreenY - ImageScreenH)) * scale);
462 int sw = (int)(Math.Min(ImageScreenW, width - ImageScreenX) * scale);
463 int sh = (int)(Math.Min(ImageScreenH, height - ImageScreenY) * scale);
464 GL.Scissor(sx, sy, Math.Max(0, sw), Math.Max(0, sh));
465 }
466
467 GL.UseProgram(_shaderProgram);
468 GL.BindVertexArray(_vao);
469
470 GL.Uniform2(_locViewportSize, (float)width, (float)height);
471 GL.Uniform1(_locTex, 0);
472 GL.ActiveTexture(TextureUnit.Texture0);
473
474 int renderedCount = 0;
475 foreach (var tile in TilesToRender)
476 {
477 if (!_textureCache.TryGetValue(tile.Index, out int texId))
478 {
479 LogDiag($"[RenderTiles] missing texture tile={tile.Index}");
480 continue;
481 }
482
483 GL.Uniform2(_locPos, tile.ScreenX, tile.ScreenY);
484 GL.Uniform2(_locSize, tile.ScreenWidth, tile.ScreenHeight);
485 GL.Uniform2(_locUvMin, tile.U0, tile.V0);
486 GL.Uniform2(_locUvMax, tile.U1, tile.V1);
487
488 LogDiag($"[RenderTiles] draw tile={tile.Index} pos=({tile.ScreenX:F1},{tile.ScreenY:F1}) size=({tile.ScreenWidth:F1},{tile.ScreenHeight:F1}) tex={texId}");
489
490 GL.BindTexture(TextureTarget.Texture2D, texId);
491 GL.DrawArrays(PrimitiveType.Triangles, 0, 6);
492 renderedCount++;
493 }
494
495 LogDiag($"[RenderTiles] rendered={renderedCount}");
496
497 GL.Disable(EnableCap.ScissorTest);
498 GL.BindVertexArray(0);
499 GL.UseProgram(0);
500 GL.Disable(EnableCap.Blend);
501 _tileFrameComplete = true;
502 }

504 private void RenderCompleteFrameFallback(int width, int height)
505 {
506 if (_completeFrameTexture == 0 || _completeFrameWidth <= 0 || _completeFrameHeight <= 0)
507 return;
508
509 GL.Disable(EnableCap.ScissorTest);
510 GL.Disable(EnableCap.Blend);
511 GL.UseProgram(_shaderProgram);
512 GL.BindVertexArray(_vao);
513 GL.Uniform2(_locViewportSize, (float)width, (float)height);
514 GL.Uniform1(_locTex, 0);
515 GL.Uniform2(_locPos, 0f, 0f);
516 GL.Uniform2(_locSize, (float)width, (float)height);
517 // Framebuffer copies use bottom-left texture origin, unlike uploaded
518 // top-down tile bytes, so reverse V while drawing the snapshot.
519 GL.Uniform2(_locUvMin, 0f, 1f);
520 GL.Uniform2(_locUvMax, 1f, 0f);
521 GL.ActiveTexture(TextureUnit.Texture0);
522 GL.BindTexture(TextureTarget.Texture2D, _completeFrameTexture);
523 GL.DrawArrays(PrimitiveType.Triangles, 0, 6);
524 GL.BindTexture(TextureTarget.Texture2D, 0);
525 GL.BindVertexArray(0);
526 GL.UseProgram(0);
527 }

529 private void CaptureCompleteTileFrame(int width, int height)
530 {
531 int physicalWidth = width * Math.Max(1, ScaleFactor);
532 int physicalHeight = height * Math.Max(1, ScaleFactor);
533 if (physicalWidth <= 0 || physicalHeight <= 0)
534 return;
535
536 if (_completeFrameTexture == 0)
537 _completeFrameTexture = GL.GenTexture();
538 if (_completeFrameTexture == 0)
539 return;
540
541 GL.BindTexture(TextureTarget.Texture2D, _completeFrameTexture);
542 GL.TexParameter(TextureTarget.Texture2D, TextureParameterName.TextureMinFilter, (int)TextureMinFilter.Linear);
543 GL.TexParameter(TextureTarget.Texture2D, TextureParameterName.TextureMagFilter, (int)TextureMagFilter.Linear);
544 GL.TexParameter(TextureTarget.Texture2D, TextureParameterName.TextureWrapS, (int)TextureWrapMode.ClampToEdge);
545 GL.TexParameter(TextureTarget.Texture2D, TextureParameterName.TextureWrapT, (int)TextureWrapMode.ClampToEdge);
546
547 if (_completeFrameWidth != physicalWidth || _completeFrameHeight != physicalHeight)
548 {
549 GL.CopyTexImage2D(TextureTarget.Texture2D, 0, InternalFormat.Rgba,
550 0, 0, physicalWidth, physicalHeight, 0);
551 _completeFrameWidth = physicalWidth;
552 _completeFrameHeight = physicalHeight;
553 }
554 else
555 {
556 GL.CopyTexSubImage2D(TextureTarget.Texture2D, 0, 0, 0,
557 0, 0, physicalWidth, physicalHeight);
558 }
559 GL.BindTexture(TextureTarget.Texture2D, 0);
560 }

574 private void CaptureWholeImageFrameIfRequested(int width, int height)
575 {
576 var callback = _pendingWholeImageCapture;
577 if (callback == null || TilesToRender.Count == 0 ||
578 !TilesToRender.All(tile => _textureCache.ContainsKey(tile.Index)))
579 return;
580
581 // Only use a frame in which the complete image fits in the viewport.
582 // A later zoomed/panned frame is not a valid overview source.
583 const float tolerance = 1.5f;
584 if (ImageScreenW <= 0 || ImageScreenH <= 0 ||
585 ImageScreenX < -tolerance || ImageScreenY < -tolerance ||
586 ImageScreenX + ImageScreenW > width + tolerance ||
587 ImageScreenY + ImageScreenH > height + tolerance)
588 return;
589
590 int scale = Math.Max(1, ScaleFactor);
591 int framebufferWidth = width * scale;
592 int framebufferHeight = height * scale;
593 int left = Math.Clamp((int)Math.Round(ImageScreenX * scale), 0, framebufferWidth - 1);
594 int top = Math.Clamp((int)Math.Round(ImageScreenY * scale), 0, framebufferHeight - 1);
595 int captureWidth = Math.Clamp((int)Math.Round(ImageScreenW * scale), 1, framebufferWidth - left);
596 int captureHeight = Math.Clamp((int)Math.Round(ImageScreenH * scale), 1, framebufferHeight - top);
597 int glBottom = framebufferHeight - top - captureHeight;
598
599 byte[] rgbaBottomUp = new byte[captureWidth * captureHeight * 4];
600 GL.ReadPixels(left, glBottom, captureWidth, captureHeight,
601 PixelFormat.Rgba, PixelType.UnsignedByte, rgbaBottomUp);
602
603 // AForge's 32-bit bitmap path expects BGRA rows from top to bottom.
604 byte[] bgraTopDown = new byte[rgbaBottomUp.Length];
605 int rowBytes = captureWidth * 4;
606 for (int y = 0; y < captureHeight; y++)
607 {
608 int sourceRow = (captureHeight - 1 - y) * rowBytes;
609 int destinationRow = y * rowBytes;
610 for (int x = 0; x < captureWidth; x++)
611 {
612 int source = sourceRow + x * 4;
613 int destination = destinationRow + x * 4;
614 bgraTopDown[destination] = rgbaBottomUp[source + 2];
615 bgraTopDown[destination + 1] = rgbaBottomUp[source + 1];
616 bgraTopDown[destination + 2] = rgbaBottomUp[source];
617 bgraTopDown[destination + 3] = 255;
618 }
619 }
620
621 _pendingWholeImageCapture = null;
622 callback(bgraTopDown, captureWidth, captureHeight);
623 }

625 private void RenderSkiaOverlay(int width, int height, int presentationFramebuffer)
626 {
627 if (OnSkiaRender == null) return;
628 InitializeSkia(presentationFramebuffer);
629 if (_skSurface == null)
630 return;
631

633 // that every cached GL binding/state may have changed so it rebinds

641 }

650 public void UploadTileTexture(TileIndex index, byte[] pixelData, int tileWidth, int tileHeight)
651 {
652 if (!_glInitialized)
653 {
654 LogDiag($"[UploadTileTexture] gl-not-initialized tile={index}");
655 return;
656 }
657
658 if (pixelData == null || pixelData.Length == 0)
659 {
660 LogDiag($"[UploadTileTexture] empty pixel data tile={index}");
661 return;
662 }
663
664 // The buffer is always BGRA = 4 bytes per pixel.
665 // GetTile pads every tile — including edge tiles — to exactly
666 // tileWidth x tileHeight pixels, so the dimensions passed by SlideRenderer
667 // (the schema tile size) always match the buffer. We just validate and bail
668 // on any genuine mismatch rather than trying to guess new dimensions.
669 if (tileWidth <= 0 || tileHeight <= 0)
670 {
671 LogDiag($"[UploadTileTexture] invalid dimensions tile={index} size={tileWidth}x{tileHeight}");
672 return;
673 }
674
675 int expectedBytes = tileWidth * tileHeight * 4;
676 if (pixelData.Length < expectedBytes)
677 {
678 LogDiag($"[UploadTileTexture] buffer too small tile={index} need={expectedBytes} got={pixelData.Length}");
679 return;
680 }
681
682 MakeCurrent();
683
684 // Check if already cached
685 if (_textureCache.ContainsKey(index))
686 {
687 LogDiag($"[UploadTileTexture] already cached tile={index}");
688 return;
689 }
690
691 long newTextureBytes = (long)tileWidth * tileHeight * 4;
692
693 // Evict old textures if the cache is full or if the new upload would
694 // push the total texture budget over its byte cap.
695 while (_textureCache.Count > 0 &&
696 (_textureCache.Count >= MAX_CACHED_TEXTURES ||
697 _textureBytes + newTextureBytes > MAX_CACHED_TEXTURE_BYTES))
698 {
699 EvictOldestTextures(1);
700 }
701
702 // Create and upload texture.
703 // Pin the managed array explicitly so the GC cannot relocate it while the
704 // driver is reading from the pointer (avoids ExecutionEngineException).
705 // PixelFormat.Bgra matches the byte order produced by ReadRegionAsync.
706 int tex = GL.GenTexture();
707 if (tex == 0)
708 {
709 Console.WriteLine($"OpenGL failed to allocate texture for tile {index}.");
710 return;
711 }
712 GL.BindTexture(TextureTarget.Texture2D, tex);
713 GL.PixelStore(PixelStoreParameter.UnpackAlignment, 4);
714
715 var handle = System.Runtime.InteropServices.GCHandle.Alloc(pixelData, System.Runtime.InteropServices.GCHandleType.Pinned);
716 try
717 {
718 GL.TexImage2D(TextureTarget.Texture2D, 0, PixelInternalFormat.Rgba,
719 tileWidth, tileHeight, 0,
720 PixelFormat.Bgra, PixelType.UnsignedByte,
721 handle.AddrOfPinnedObject());
722 }
723 finally
724 {
725 handle.Free();
726 }
727
728 GL.TexParameter(TextureTarget.Texture2D, TextureParameterName.TextureMinFilter, (int)TextureMinFilter.Linear);

820 private void EvictOldestTextures(int count)
821 {
822 // Prefer off-screen textures. Tiles in the displayed viewport and
823 // incoming tiles for a pending level switch must remain resident long
824 // enough to form a complete frame.
825 var toRemove = _textureCache
826 .Where(kvp => !_protectedTileIndices.Contains(kvp.Key))
827 .Take(count)
828 .ToList();
829
830 // This should only be needed if a viewport itself exceeds the cache.
831 if (toRemove.Count < count)
832 {
833 var selected = new HashSet<TileIndex>(toRemove.Select(kvp => kvp.Key));
834 toRemove.AddRange(_textureCache
835 .Where(kvp => !selected.Contains(kvp.Key))
836 .Take(count - toRemove.Count));
837 }
838 foreach (var kvp in toRemove)
839 {
840 GL.DeleteTexture(kvp.Value);
841 if (_textureSizes.TryGetValue(kvp.Key, out var sz))
842 {
843 _textureBytes -= (long)sz.w * sz.h * 4;
844 _textureSizes.Remove(kvp.Key);
845 }
846 _textureCache.Remove(kvp.Key);
847 }
