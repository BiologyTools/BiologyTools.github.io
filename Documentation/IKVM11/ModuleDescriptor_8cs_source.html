42 public static ModuleDescriptor Read(ClassFile clazz)
43 {
44 if (clazz is null)
45 throw new ArgumentNullException(nameof(clazz));
46
47 try
48 {
49 if (clazz.Version < 53)
50 throw new InvalidModuleDescriptorException($"Unsupported class file version '{clazz.Version}'.");
51
52 if ((clazz.AccessFlags & AccessFlag.Module) != AccessFlag.Module)
53 throw new InvalidModuleDescriptorException("AccessFlags should be ACC_MODULE.");
54
55 var clazzName = clazz.Constants.Get(clazz.This).Name;
56 if (clazzName != "module-info")
57 throw new InvalidModuleDescriptorException("Class name should be 'module-info'.");
58
59 if (clazz.Super.IsNotNil)
60 throw new InvalidModuleDescriptorException("Bad super class.");
61
62 if (clazz.Fields.Count > 0)
63 throw new InvalidModuleDescriptorException("Bad fields.");
64
65 if (clazz.Methods.Count > 0)
66 throw new InvalidModuleDescriptorException("Bad methods.");
67
68 if (clazz.Interfaces.Count > 0)
69 throw new InvalidModuleDescriptorException("Bad interfaces.");
70
71 var builder = ReadModuleAttribute(clazz);
72 ReadModulePackagesAttribute(clazz, builder);
73 ReadModuleMainClassAttribute(clazz, builder);
74
75 var descriptor = builder.Build();
76
77 // various checks based on whether we are reading java.base or not
78 if (descriptor.Name == "java.base")
79 {
80 // building java.base itself: should have no requires
81 if (descriptor.Requires.Count > 0)
82 throw new InvalidModuleDescriptorException("The requires table for java.base must be 0 length.");
83 }
84 else
85 {
86 // modules must explicitely require java.base
87 if (descriptor.Requires.Any(i => i.Name == "java.base") == false)
88 throw new InvalidModuleDescriptorException("The requires table must have an entry for java.base.");
89 }
90
91 return descriptor;
92 }
93 catch (ByteCodeException e)
94 {
95 throw new InvalidModuleDescriptorException("Unable to read module descriptor.", e);
96 }
97 }

121 static Builder ReadModuleAttribute(ClassFile clazz, in ModuleAttribute attribute)
122 {
123 var moduleName = clazz.Constants.Get(attribute.Name).Name;
124 if (moduleName is not null)
125 moduleName = DecodeModuleName(moduleName);
126 if (moduleName is null || string.IsNullOrEmpty(moduleName))
127 throw new InvalidModuleDescriptorException("Module name not found.");
128
129 var modifiers = default(ModuleFlag);
130 if ((attribute.Flags & ModuleFlag.Open) != 0)
131 modifiers |= ModuleFlag.Open;
132 if ((attribute.Flags & ModuleFlag.Synthetic) != 0)
133 modifiers |= ModuleFlag.Synthetic;
134 if ((attribute.Flags & ModuleFlag.Mandated) != 0)
135 modifiers |= ModuleFlag.Mandated;
136
137 var builder = CreateModule(moduleName, modifiers);
138
139 if (attribute.Version.IsNotNil)
140 builder = builder.Version(clazz.Constants.Get(attribute.Version).Value);
141
142 ReadModuleRequires(clazz, attribute, builder);
143 ReadModuleExports(clazz, attribute, builder);
144 ReadModuleOpens(clazz, attribute, builder);
145 ReadModuleUses(clazz, attribute, builder);
146 ReadModuleProvides(clazz, attribute, builder);
147
148 return builder;

223 static void ReadModuleRequires(ClassFile clazz, in ModuleAttribute attribute, Builder builder)
224 {
225 foreach (var requires in attribute.Requires)
226 {
227 var moduleName = clazz.Constants.Get(requires.Module).Name;
228 if (moduleName is not null)
229 moduleName = DecodeModuleName(moduleName);
230 if (moduleName is null || string.IsNullOrEmpty(moduleName))
231 throw new InvalidModuleDescriptorException("Bad module name on requires.");
232
233 var modifiers = default(ModuleRequiresFlag);
234 if ((requires.Flag & ModuleRequiresFlag.Transitive) != 0)
235 modifiers |= ModuleRequiresFlag.Transitive;
236 if ((requires.Flag & ModuleRequiresFlag.StaticPhase) != 0)
237 modifiers |= ModuleRequiresFlag.StaticPhase;
238 if ((requires.Flag & ModuleRequiresFlag.Synthetic) != 0)
239 modifiers |= ModuleRequiresFlag.Synthetic;
240 if ((requires.Flag & ModuleRequiresFlag.Mandated) != 0)
241 modifiers |= ModuleRequiresFlag.Mandated;
242
243 if (moduleName == "java.base")
244 {
245 if ((modifiers & ModuleRequiresFlag.Synthetic) != 0)
246 throw new InvalidModuleDescriptorException("The requires entry for java.base has ACC_SYNTHETIC set");
247
248 // requires transitive java.base is illegal unless:
249 // - the major version is 53 (JDK 9), or:
250 // - the classfile is a preview classfile, or:
251 // - the module is deemed to be participating in preview
252 // (i.e. the module is a java.* module)
253 // requires static java.base is illegal unless:
254 // - the major version is 53 (JDK 9), or:
255 if (clazz.Version.Major >= 54)
256 {
257 var hasTransitive = (modifiers & ModuleRequiresFlag.Transitive) != 0;
258 var hasStatic = (modifiers & ModuleRequiresFlag.StaticPhase) != 0;
259 if ((hasTransitive && "java.se" != moduleName) || hasStatic)
260 throw new InvalidModuleDescriptorException($"The requires entry for java.base has {(hasStatic ? "ACC_STATIC_PHASE" : "ACC_TRANSITIVE")} set.");
261 }
262 }
263
264 if (requires.Version.IsNotNil)
265 {
266 var version = clazz.Constants.Get(requires.Version).Value;
267 builder = builder.Requires(modifiers, moduleName, version);
268 }
269 else
270 {
271 builder = builder.Requires(modifiers, moduleName);
272 }
273 }
274 }

382 static string DecodeModuleName(string value)
383 {
384 if (value.Length == 0)
385 throw new InvalidModuleDescriptorException("Module name is zero length.");
386
387 int i = 0;
388 while (i < value.Length)
389 {
390 int cp = char.ConvertToUtf32(value, i);
391 if (cp == ':' || cp == '@' || cp < 0x20)
392 throw new InvalidModuleDescriptorException($"Module name has illegal character: U+{cp:X4}.");
393
394 // blackslash is the escape character
395 if (cp == '\\')
396 return DecodeModuleName(i, value);
397
398 i += CharCount(cp);
399 }
400
401 return value;
402 }

408 static string DecodeModuleName(int i, string value)
409 {
410 var sb = new ValueStringBuilder(value.Length - i);
411
412 // copy the code points that have been checked
413 int j = 0;
414 while (j < i)
415 {
416 int cp = char.ConvertToUtf32(value, j);
417 AppendUtf32(ref sb, cp);
418 j += CharCount(cp);
419 }
420
421 // decode from position i to end
422 while (i < value.Length)
423 {
424 int cp = char.ConvertToUtf32(value, i);
425 if (cp == ':' || cp == '@' || cp < 0x20)
426 {
427 throw new InvalidModuleDescriptorException($"Module name has illegal character: U+{cp:X4}.");
428 }
429
430 // blackslash is the escape character
431 if (cp == '\\')
432 {
433 j = i + CharCount(cp);
434 if (j >= value.Length)
435 throw new InvalidModuleDescriptorException("Module name has illegal escape sequence.");
436
437 int next = char.ConvertToUtf32(value, j);
438 if (next != '\\' && next != ':' && next != '@')
439 throw new InvalidModuleDescriptorException("Module name has illegal escape sequence.");
440
441 AppendUtf32(ref sb, next);
442 i += CharCount(next);
443 }
444 else
445 {
446 AppendUtf32(ref sb, cp);
447 }
448
449 i += CharCount(cp);
450 }
451
452 return sb.ToString();
453 }

460 static void AppendUtf32(ref ValueStringBuilder builder, int utf32)

464 builder.Append(c[..l]);

473 static int WriteCodePoint(int utf32, Span<char> utf16)
474 {
475 if (utf32 < 0xD800 || (utf32 > 0xDFFF && utf32 < 0x10000))
476 {
477 utf16[0] = (char)utf32;
478 utf16[1] = (char)0;
479 return 1;
480 }
481
482 utf32 -= 0x010000;
483
484 utf16[0] = (char)(((0b11111111110000000000 & utf32) >> 10) + 0xD800);
485 utf16[1] = (char)(((0b00000000001111111111 & utf32) >> 00) + 0xDC00);
486
487 return 2;
488 }
