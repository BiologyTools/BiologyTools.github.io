119 void Read(Stream stream, bool mapped)
120 {
121 var br = new BinaryReader(stream);
122 peFile.Read(br, mapped);
123 stream.Seek(peFile.RvaToFileOffset(peFile.GetComDescriptorVirtualAddress()), SeekOrigin.Begin);
124 cliHeader.Read(br);
125
126 stream.Seek(peFile.RvaToFileOffset(cliHeader.MetaData.VirtualAddress), SeekOrigin.Begin);
127 foreach (var sh in ReadStreamHeaders(br, out imageRuntimeVersion))
128 {
129 switch (sh.Name)
130 {
131 case "#Strings":
132 stringHeap = ReadHeap(stream, sh.Offset, sh.Size);
133 break;
134 case "#Blob":
135 blobHeap = ReadHeap(stream, sh.Offset, sh.Size);
136 break;
137 case "#US":
138 userStringHeapOffset = sh.Offset;
139 userStringHeapSize = sh.Size;
140 break;
141 case "#GUID":
142 guidHeap = ReadHeap(stream, sh.Offset, sh.Size);
143 break;
144 case "#~":
145 case "#-":
146 stream.Seek(peFile.RvaToFileOffset(cliHeader.MetaData.VirtualAddress + sh.Offset), SeekOrigin.Begin);
147 ReadTables(br);
148 break;
149 default:
150 // we ignore unknown streams, because the CLR does so too
151 // (and some obfuscators add bogus streams)
152 break;
153 }
154 }

162 static StreamHeader[] ReadStreamHeaders(BinaryReader br, out string version)
163 {
164 var signature = br.ReadUInt32();
165 if (signature != 0x424A5342)
166 throw new BadImageFormatException("Invalid metadata signature");
167
168 /*ushort MajorVersion =*/
169 br.ReadUInt16();
170 /*ushort MinorVersion =*/
171 br.ReadUInt16();
172 /*uint Reserved =*/
173 br.ReadUInt32();
174 var Length = br.ReadUInt32();
175 var buf = br.ReadBytes((int)Length);
176 version = Encoding.UTF8.GetString(buf).TrimEnd('\u0000');
177 /*ushort Flags =*/
178 br.ReadUInt16();
179
180 var streams = br.ReadUInt16();
181 var streamHeaders = new StreamHeader[streams];
182 for (int i = 0; i < streamHeaders.Length; i++)
183 {
184 streamHeaders[i] = new StreamHeader();
185 streamHeaders[i].Read(br);
186 }
187
188 return streamHeaders;
189 }
190
191 void ReadTables(BinaryReader br)
192 {
193 var tables = GetTables();
194
195 /*uint Reserved0 =*/
196 br.ReadUInt32();
197 var majorVersion = br.ReadByte();
198 var minorVersion = br.ReadByte();
199 metadataStreamVersion = majorVersion << 16 | minorVersion;
200 var heapSizes = br.ReadByte();
201 /*byte Reserved7 =*/
202 br.ReadByte();
203
204 ulong valid = br.ReadUInt64();
205 ulong sorted = br.ReadUInt64();
206 for (int i = 0; i < 64; i++)
207 {
208 if ((valid & (1UL << i)) != 0)
209 {
210 tables[i].Sorted = (sorted & (1UL << i)) != 0;
211 tables[i].RowCount = br.ReadInt32();
212 }
213 }
214
215 var mr = new MetadataReader(this, br.BaseStream, heapSizes);
216 for (int i = 0; i < 64; i++)
217 if ((valid & (1UL << i)) != 0)
218 tables[i].Read(mr);
219
220 if (ParamPtrTable.RowCount != 0)
221 throw new NotImplementedException("ParamPtr table support has not yet been implemented.");
222 }

224 byte[] ReadHeap(Stream stream, uint offset, uint size)
225 {
226 var buf = new byte[size];
227 stream.Seek(peFile.RvaToFileOffset(cliHeader.MetaData.VirtualAddress + offset), SeekOrigin.Begin);
228
229 for (var pos = 0; pos < buf.Length;)
230 {
231 var read = stream.Read(buf, pos, buf.Length - pos);
232 if (read == 0)
233 throw new BadImageFormatException();
234
235 pos += read;
236 }
237
238 return buf;
239 }

288 internal override string GetString(StringHandle handle)
289 {
290 if (handle.IsNil)
291 return null;
292
293 if (!strings.TryGetValue(handle, out var str))
294 {
295 int len = 0;
296 while (stringHeap[MetadataTokens.GetHeapOffset(handle) + len] != 0)
297 len++;
298
299 str = Encoding.UTF8.GetString(stringHeap, MetadataTokens.GetHeapOffset(handle), len);
300 strings.Add(handle, str);
301 }
302
303 return str;
304 }

306 static int ReadCompressedUInt(byte[] buffer, ref int offset)
307 {
308 var b1 = buffer[offset++];
309 if (b1 <= 0x7F)
310 {
311 return b1;
312 }
313 else if ((b1 & 0xC0) == 0x80)
314 {
315 var b2 = buffer[offset++];
316 return ((b1 & 0x3F) << 8) | b2;
317 }
318 else
319 {
320 var b2 = buffer[offset++];
321 var b3 = buffer[offset++];
322 var b4 = buffer[offset++];
323 return ((b1 & 0x3F) << 24) + (b2 << 16) + (b3 << 8) + b4;
324 }
325 }

327 internal byte[] GetBlobCopy(BlobHandle handle)
328 {
329 var idx = MetadataTokens.GetHeapOffset(handle);
330 var len = ReadCompressedUInt(blobHeap, ref idx);
331 var buf = new byte[len];
332 Buffer.BlockCopy(blobHeap, idx, buf, 0, len);
333 return buf;
334 }

341 public override string ResolveString(int metadataToken)
342 {
343 if ((metadataToken >> 24) != 0x70)
344 throw TokenOutOfRangeException(metadataToken);
345
346 var h = MetadataTokens.StringHandle(metadataToken);
347
348 if (strings.TryGetValue(h, out var str) == false)
349 {
350 lazyUserStringHeap ??= ReadHeap(GetStream(), userStringHeapOffset, userStringHeapSize);
351
352 var index = metadataToken & 0xFFFFFF;
353 var len = ReadCompressedUInt(lazyUserStringHeap, ref index) & ~1;
354 var sb = new StringBuilder(len / 2);
355 for (int i = 0; i < len; i += 2)
356 {
357 var ch = (char)(lazyUserStringHeap[index + i] | lazyUserStringHeap[index + i + 1] << 8);
358 sb.Append(ch);
359 }
360
361 str = sb.ToString();
362 strings.Add(h, str);
363 }
364
365 return str;
366 }

368 internal override Type ResolveType(int metadataToken, IGenericContext context)
369 {
370 int index = (metadataToken & 0xFFFFFF) - 1;
371 if (index < 0)
372 throw TokenOutOfRangeException(metadataToken);
373
374 if ((metadataToken >> 24) == TypeDefTable.Index && index < TypeDefTable.RowCount)
375 {
376 PopulateTypeDef();
377 return typeDefs[index];
378 }
379
380 if ((metadataToken >> 24) == TypeRefTable.Index && index < TypeRefTable.RowCount)
381 {
382 typeRefs ??= new Type[TypeRefTable.records.Length];
383
384 if (typeRefs[index] == null)
385 {
386 var scope = TypeRefTable.records[index].ResolutionScope;
387 switch (scope >> 24)
388 {
389 case AssemblyRefTable.Index:
390 {
391 var assembly = ResolveAssemblyRef((scope & 0xFFFFFF) - 1);
392 var typeName = GetTypeName(TypeRefTable.records[index].TypeNamespace, TypeRefTable.records[index].TypeName);
393 typeRefs[index] = assembly.ResolveType(this, typeName);
394 break;
395 }
396 case TypeRefTable.Index:
397 {
398 var outer = ResolveType(scope, null);
399 var typeName = GetTypeName(TypeRefTable.records[index].TypeNamespace, TypeRefTable.records[index].TypeName);
400 typeRefs[index] = outer.ResolveNestedType(this, typeName);
401 break;
402 }
403 case ModuleTable.Index:
404 case ModuleRefTable.Index:
405 {
406 Module module;
407
408 if (scope >> 24 == ModuleTable.Index)
409 {
410 if (scope == 0 || scope == 1)
411 module = this;
412 else
413 throw new NotImplementedException("self reference scope?");
414 }
415 else
416 {
417 module = ResolveModuleRef(ModuleRefTable.records[(scope & 0xFFFFFF) - 1]);
418 }
419
420 var typeName = GetTypeName(TypeRefTable.records[index].TypeNamespace, TypeRefTable.records[index].TypeName);
421 typeRefs[index] = module.FindType(typeName) ?? module.Universe.GetMissingTypeOrThrow(this, module, null, typeName);
422 break;
423 }
424 default:
425 throw new NotImplementedException("ResolutionScope = " + scope.ToString("X"));
426 }
427 }
428
429 return typeRefs[index];
430 }
431
432 if ((metadataToken >> 24) == TypeSpecTable.Index && index < TypeSpecTable.RowCount)
433 {
434 typeSpecs ??= new Type[TypeSpecTable.records.Length];
435
436 var type = typeSpecs[index];
437 if (type == null)
438 {
439 var tc = context == null ? null : new TrackingGenericContext(context);
440 typeSpecs[index] = MarkerType.LazyResolveInProgress;
441
442 try
443 {
444 type = Signature.ReadTypeSpec(this, ByteReader.FromBlob(blobHeap, TypeSpecTable.records[index]), tc);
445 }
446 finally
447 {
448 typeSpecs[index] = null;
449 }
450
451 if (tc == null || !tc.IsUsed)
452 typeSpecs[index] = type;
453 }
454 else if (type == MarkerType.LazyResolveInProgress)
455 {
456 if (Universe.MissingMemberResolution)
457 {
458 return Universe
459 .GetMissingTypeOrThrow(this, this, null, new TypeName(null, "Cyclic TypeSpec " + metadataToken.ToString("X")))
460 .SetCyclicTypeSpec()
461 .SetMetadataTokenForMissing(metadataToken, 0);
462 }
463
464 throw new BadImageFormatException("Cyclic TypeSpec " + metadataToken.ToString("X"));
465 }
466
467 return type;
468 }
469
470 throw TokenOutOfRangeException(metadataToken);
471 }

549 static Guid GuidFromSpan(ReadOnlySpan<byte> b)
550 {
551#if NETFRAMEWORK
552 var _a = (b[3] << 24) | (b[2] << 16) | (b[1] << 8) | b[0];
553 var _b = (short)((b[5] << 8) | b[4]);
554 var _c = (short)((b[7] << 8) | b[6]);
555 var _d = b[8];
556 var _e = b[9];
557 var _f = b[10];
558 var _g = b[11];
559 var _h = b[12];
560 var _i = b[13];
561 var _j = b[14];
562 var _k = b[15];
563 return new Guid(_a, _b, _c, _d, _e, _f, _g, _h, _i, _j, _k);
564#else
565 return new Guid(b);
566#endif

606 public override MemberInfo ResolveMember(int metadataToken, Type[] genericTypeArguments, Type[] genericMethodArguments)
607 {
608 switch (metadataToken >> 24)
609 {
610 case FieldTable.Index:
611 return ResolveField(metadataToken, genericTypeArguments, genericMethodArguments);
612 case MemberRefTable.Index:
613 int index = (metadataToken & 0xFFFFFF) - 1;
614 if (index < 0 || index >= MemberRefTable.RowCount)
615 goto default;
616
617 return GetMemberRef(index, genericTypeArguments, genericMethodArguments);
618 case MethodDefTable.Index:
619 case MethodSpecTable.Index:
620 return ResolveMethod(metadataToken, genericTypeArguments, genericMethodArguments);
621 case TypeRefTable.Index:
622 case TypeDefTable.Index:
623 case TypeSpecTable.Index:
624 return ResolveType(metadataToken, genericTypeArguments, genericMethodArguments);
625 default:
626 throw TokenOutOfRangeException(metadataToken);
627 }
628 }

691 public override MethodBase ResolveMethod(int metadataToken, Type[] genericTypeArguments, Type[] genericMethodArguments)
692 {
693 int index = (metadataToken & 0xFFFFFF) - 1;
694 if (index < 0)
695 {
696 throw TokenOutOfRangeException(metadataToken);
697 }
698 else if ((metadataToken >> 24) == MethodDefTable.Index && index < MethodDefTable.RowCount)
699 {
700 return GetMethodAt(null, index);
701 }
702 else if ((metadataToken >> 24) == MemberRefTable.Index && index < MemberRefTable.RowCount)
703 {
704 var method = GetMemberRef(index, genericTypeArguments, genericMethodArguments) as MethodBase;
705 if (method != null)
706 return method;
707
708 throw new ArgumentException(String.Format("Token 0x{0:x8} is not a valid MethodBase token in the scope of module {1}.", metadataToken, this.Name), "metadataToken");
709 }
710 else if ((metadataToken >> 24) == MethodSpecTable.Index && index < MethodSpecTable.RowCount)
711 {
712 var method = (MethodInfo)ResolveMethod(MethodSpecTable.records[index].Method, genericTypeArguments, genericMethodArguments);
713 var instantiation = ByteReader.FromBlob(blobHeap, MethodSpecTable.records[index].Instantiation);
714 return method.MakeGenericMethod(Signature.ReadMethodSpec(this, instantiation, new GenericContext(genericTypeArguments, genericMethodArguments)));
715 }
716 else
717 {
718 throw TokenOutOfRangeException(metadataToken);
719 }
720 }

779 MemberInfo GetMemberRef(int index, Type[] genericTypeArguments, Type[] genericMethodArguments)
780 {
781 memberRefs ??= new MemberInfo[MemberRefTable.records.Length];
782
783 if (memberRefs[index] == null)
784 {
785 var owner = MemberRefTable.records[index].Class;
786 var sig = MemberRefTable.records[index].Signature;
787 var name = GetString(MemberRefTable.records[index].Name);
788 switch (owner >> 24)
789 {
790 case MethodDefTable.Index:
791 return GetMethodAt(null, (owner & 0xFFFFFF) - 1);

806 }
