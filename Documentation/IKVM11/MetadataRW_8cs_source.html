93 static bool IsBig(int bitsUsed, params Table[] tables)
94 {
95 var limit = 1 << (16 - bitsUsed);
96 foreach (var table in tables)
97 if (table.RowCount >= limit)
98 return true;
99
100 return false;
101 }
