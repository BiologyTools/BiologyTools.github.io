213 static void ResolveMethod(global::java.lang.invoke.MemberName self, global::java.lang.Class caller)
214 {
215 bool invokeSpecial = self.getReferenceKind() == global::java.lang.invoke.MethodHandleNatives.Constants.REF_invokeSpecial;
216 bool newInvokeSpecial = self.getReferenceKind() == global::java.lang.invoke.MethodHandleNatives.Constants.REF_newInvokeSpecial;
217 bool searchBaseClasses = !newInvokeSpecial;
218 RuntimeJavaMethod mw = RuntimeJavaType.FromClass(self.getDeclaringClass()).GetMethod(self.getName(), self.getSignature().Replace('/', '.'), searchBaseClasses);
219 if (mw == null)
220 {
221 if (self.getReferenceKind() == global::java.lang.invoke.MethodHandleNatives.Constants.REF_invokeInterface)
222 {
223 mw = RuntimeJavaType.FromClass(self.getDeclaringClass()).GetInterfaceMethod(self.getName(), self.getSignature().Replace('/', '.'));
224 if (mw == null)
225 {
226 mw = JVM.Context.JavaBase.TypeOfJavaLangObject.GetMethod(self.getName(), self.getSignature().Replace('/', '.'), false);
227 }
228 if (mw != null && mw.IsConstructor)
229 {
230 throw new global::java.lang.IncompatibleClassChangeError("Found interface " + self.getDeclaringClass().getName() + ", but class was expected");
231 }
232 }
233 if (mw == null)
234 {
235 string msg = String.Format(invokeSpecial ? "{0}: method {1}{2} not found" : "{0}.{1}{2}", self.getDeclaringClass().getName(), self.getName(), self.getSignature());
236 throw new global::java.lang.NoSuchMethodError(msg);
237 }
238 }
239 if (mw.IsStatic != IsReferenceKindStatic(self.getReferenceKind()))
240 {
241 string msg = String.Format(mw.IsStatic ? "Expecting non-static method {0}.{1}{2}" : "Expected static method {0}.{1}{2}", mw.DeclaringType.Name, self.getName(), self.getSignature());
242 throw new global::java.lang.IncompatibleClassChangeError(msg);
243 }
244 if (self.getReferenceKind() == global::java.lang.invoke.MethodHandleNatives.Constants.REF_invokeVirtual && mw.DeclaringType.IsInterface)
245 {
246 throw new global::java.lang.IncompatibleClassChangeError("Found interface " + mw.DeclaringType.Name + ", but class was expected");
247 }
248 if (!mw.IsPublic && self.getReferenceKind() == global::java.lang.invoke.MethodHandleNatives.Constants.REF_invokeInterface)
249 {
250 throw new global::java.lang.IncompatibleClassChangeError("private interface method requires invokespecial, not invokeinterface: method " + self.getDeclaringClass().getName() + "." + self.getName() + self.getSignature());
251 }
252 if (mw.IsConstructor && mw.DeclaringType == JVM.Context.JavaBase.TypeOfJavaLangString)
253 {
254 self.vmtarget = CreateMemberNameDelegate(mw, caller, false, self.getMethodType().changeReturnType(JVM.Context.JavaBase.TypeOfJavaLangString.ClassObject));
255 }
256 else if (!mw.IsConstructor || invokeSpecial || newInvokeSpecial)
257 {
258 global::java.lang.invoke.MethodType methodType = self.getMethodType();
259 if (!mw.IsStatic)
260 {
261 methodType = methodType.insertParameterTypes(0, mw.DeclaringType.ClassObject);
262 if (newInvokeSpecial)
263 {
264 methodType = methodType.changeReturnType(global::java.lang.Void.TYPE);
265 }
266 }
267 self.vmtarget = CreateMemberNameDelegate(mw, caller, self.hasReceiverTypeDispatch(), methodType);
268 }
269 SetModifiers(self, mw);
270 self._flags(self._flags() | (mw.IsConstructor ? global::java.lang.invoke.MethodHandleNatives.Constants.MN_IS_CONSTRUCTOR : global::java.lang.invoke.MethodHandleNatives.Constants.MN_IS_METHOD));
271 if (self.getReferenceKind() == global::java.lang.invoke.MethodHandleNatives.Constants.REF_invokeVirtual && (mw.IsPrivate || mw.IsFinal || mw.IsConstructor))
272 {
273 int flags = self._flags();
274 flags -= global::java.lang.invoke.MethodHandleNatives.Constants.REF_invokeVirtual << global::java.lang.invoke.MethodHandleNatives.Constants.MN_REFERENCE_KIND_SHIFT;
275 flags += global::java.lang.invoke.MethodHandleNatives.Constants.REF_invokeSpecial << global::java.lang.invoke.MethodHandleNatives.Constants.MN_REFERENCE_KIND_SHIFT;
276 self._flags(flags);
277 }
278 if (mw.HasCallerID || RuntimeByteCodeJavaType.RequiresDynamicReflectionCallerClass(mw.DeclaringType.Name, mw.Name, mw.Signature))
279 {
280 self._flags(self._flags() | global::java.lang.invoke.MemberName.CALLER_SENSITIVE);
281 }
282 }
