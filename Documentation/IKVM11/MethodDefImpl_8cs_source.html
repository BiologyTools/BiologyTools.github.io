96 void PopulateParameters()
97 {
98 if (parameters == null)
99 {
100 var methodSignature = MethodSignature;
101 parameters = new ParameterInfo[methodSignature.GetParameterCount()];
102
103 var parameter = module.MethodDefTable.records[index].ParamList - 1;
104 var end = module.MethodDefTable.records.Length > index + 1 ? module.MethodDefTable.records[index + 1].ParamList - 1 : module.ParamTable.records.Length;
105 for (; parameter < end; parameter++)
106 {
107 var seq = module.ParamTable.records[parameter].Sequence - 1;
108 if (seq == -1)
109 returnParameter = new ParameterInfoImpl(this, seq, parameter);
110 else
111 parameters[seq] = new ParameterInfoImpl(this, seq, parameter);
112 }
113
114 for (int i = 0; i < parameters.Length; i++)
115 if (parameters[i] == null)
116 parameters[i] = new ParameterInfoImpl(this, i, -1);
117
118 if (returnParameter == null)
119 returnParameter = new ParameterInfoImpl(this, -1, -1);
120 }
121 }

189 void PopulateGenericArguments()
190 {
191 if (typeArgs == null)
192 {
193 var token = MetadataToken;
194 var first = module.GenericParamTable.FindFirstByOwner(token);
195 if (first == -1)
196 {
197 typeArgs = Type.EmptyTypes;
198 }
199 else
200 {
201 var list = new List<Type>();
202 var len = module.GenericParamTable.records.Length;
203 for (int i = first; i < len && module.GenericParamTable.records[i].Owner == token; i++)
204 list.Add(new GenericTypeParameter(module, i, Signature.ELEMENT_TYPE_MVAR));
205
206 typeArgs = list.ToArray();
207 }
208 }

256 {
257 typeArgs ??= declaringType.GetGenericArguments();
258 list ??= new List<MethodInfo>();
259 list.Add((MethodInfo)module.ResolveMethod(module.MethodImplTable.records[i].MethodDeclaration, typeArgs, null));
