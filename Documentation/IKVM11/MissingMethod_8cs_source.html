51 MethodInfo TryGetForwarder()
52 {
53 if (forwarder == null && !declaringType.__IsMissing)
54 {
55 var mb = declaringType.FindMethod(name, signature);
56 var ci = mb as ConstructorInfo;
57 if (ci != null)
58 forwarder = ci.GetMethodInfo();
59 else
60 forwarder = (MethodInfo)mb;
61 }
