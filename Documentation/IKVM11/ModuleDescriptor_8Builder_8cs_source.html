118 public Builder Requires(ModuleRequires requires)
119 {
120 if (_automatic)
121 throw new InvalidOperationException("Automatic modules cannot declare dependences");
122 if (_name.Equals(requires.Name))
123 throw new ArgumentException("Dependence on self.");
124 if (_requiresKeys != null && _requiresKeys.Contains(requires.Name))
125 throw new InvalidOperationException($"Dependence upon {requires.Name} already declared.");
126
127 _requiresKeys ??= new HashSet<string>();
128 _requiresKeys.Add(requires.Name);
129 _requires ??= ImmutableHashSet.CreateBuilder<ModuleRequires>();
130 _requires.Add(requires);
