31 private static ByteCodeMetaData[] data = new ByteCodeMetaData[256];

38 private ByteCodeMetaData(ByteCode bc, ByteCodeMode reg, ByteCodeModeWide wide, bool cannotThrow)
39 {
40 this.reg = reg;
41 this.wide = wide;
42 this.normbc = (NormalizedByteCode)bc;
43 this.arg = 0;
44 this.flags = ByteCodeFlags.None;
45 if (cannotThrow)
46 {
47 this.flags |= ByteCodeFlags.CannotThrow;
48 }
49 data[(int)bc] = this;
50 }

80 internal static NormalizedByteCode GetNormalizedByteCode(ByteCode bc)
81 {
82 return data[(int)bc].normbc;
83 }
84
85 internal static int GetArg(ByteCode bc, int arg)
86 {
87 if ((data[(int)bc].flags & ByteCodeFlags.FixedArg) != 0)
88 {
89 return data[(int)bc].arg;
90 }
91 return arg;
92 }

94 internal static ByteCodeMode GetMode(ByteCode bc)

96 return data[(int)bc].reg;
97 }
98
99 internal static ByteCodeModeWide GetWideMode(ByteCode bc)
100 {
101 return data[(int)bc].wide;
102 }

104 internal static ByteCodeFlowControl GetFlowControl(NormalizedByteCode bc)
105 {
106 switch (bc)
107 {
108 case NormalizedByteCode.__tableswitch:
109 case NormalizedByteCode.__lookupswitch:
110 return ByteCodeFlowControl.Switch;
111
112 case NormalizedByteCode.__goto:
113 case NormalizedByteCode.__goto_finally:
114 return ByteCodeFlowControl.Branch;
115
116 case NormalizedByteCode.__ifeq:
117 case NormalizedByteCode.__ifne:
118 case NormalizedByteCode.__iflt:
119 case NormalizedByteCode.__ifge:
120 case NormalizedByteCode.__ifgt:
121 case NormalizedByteCode.__ifle:
122 case NormalizedByteCode.__if_icmpeq:
123 case NormalizedByteCode.__if_icmpne:
124 case NormalizedByteCode.__if_icmplt:
125 case NormalizedByteCode.__if_icmpge:
126 case NormalizedByteCode.__if_icmpgt:
127 case NormalizedByteCode.__if_icmple:
128 case NormalizedByteCode.__if_acmpeq:
129 case NormalizedByteCode.__if_acmpne:
130 case NormalizedByteCode.__ifnull:
131 case NormalizedByteCode.__ifnonnull:
132 return ByteCodeFlowControl.CondBranch;
133
134 case NormalizedByteCode.__ireturn:
135 case NormalizedByteCode.__lreturn:
136 case NormalizedByteCode.__freturn:
137 case NormalizedByteCode.__dreturn:
138 case NormalizedByteCode.__areturn:
139 case NormalizedByteCode.__return:
140 return ByteCodeFlowControl.Return;
141
142 case NormalizedByteCode.__athrow:
143 case NormalizedByteCode.__athrow_no_unmap:
144 case NormalizedByteCode.__static_error:
145 return ByteCodeFlowControl.Throw;
146
147 default:
148 return ByteCodeFlowControl.Next;
149 }
150 }
151
152 internal static bool CanThrowException(NormalizedByteCode bc)
153 {
154 switch (bc)
155 {
156 case NormalizedByteCode.__dynamic_invokeinterface:
157 case NormalizedByteCode.__dynamic_invokestatic:
158 case NormalizedByteCode.__dynamic_invokevirtual:
159 case NormalizedByteCode.__dynamic_getstatic:
160 case NormalizedByteCode.__dynamic_putstatic:
161 case NormalizedByteCode.__dynamic_getfield:
162 case NormalizedByteCode.__dynamic_putfield:
163 case NormalizedByteCode.__clone_array:
164 case NormalizedByteCode.__static_error:
165 case NormalizedByteCode.__methodhandle_invoke:
166 case NormalizedByteCode.__methodhandle_link:
167 return true;
168 case NormalizedByteCode.__iconst:
169 case NormalizedByteCode.__ldc_nothrow:
170 return false;
171 default:
172 return (data[(int)bc].flags & ByteCodeFlags.CannotThrow) == 0;
173 }
174 }
175
176 internal static bool IsBranch(NormalizedByteCode bc)
177 {
178 switch (data[(int)bc].reg)
179 {
180 case ByteCodeMode.Branch_2:
181 case ByteCodeMode.Branch_4:
182 case ByteCodeMode.Lookupswitch:
183 case ByteCodeMode.Tableswitch:
184 return true;
185 default:
186 return false;
187 }
188 }
