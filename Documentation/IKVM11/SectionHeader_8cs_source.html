54 internal void Read(BinaryReader br)
55 {
56 var name = new char[8];
57 int len = 8;
58 for (int i = 0; i < 8; i++)
59 {
60 var b = br.ReadByte();
61 name[i] = (char)b;
62 if (b == 0 && len == 8)
63 len = i;
64 }
65
66 Name = new string(name, 0, len);
67 VirtualSize = br.ReadUInt32();
68 VirtualAddress = br.ReadUInt32();
