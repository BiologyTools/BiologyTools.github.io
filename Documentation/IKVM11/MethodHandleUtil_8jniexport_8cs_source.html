82 internal java.lang.invoke.MethodType GetDelegateMethodType(Type type)
83 {
84 java.lang.Class[] types;
85 var mi = GetDelegateInvokeMethod(type);
86 var pi = mi.GetParameters();
87 if (pi.Length > 0 && IsPackedArgsContainer(pi[pi.Length - 1].ParameterType))
88 {
89 var list = new System.Collections.Generic.List<java.lang.Class>();
90 for (int i = 0; i < pi.Length - 1; i++)
91 list.Add(context.ClassLoaderFactory.GetJavaTypeFromType(pi[i].ParameterType).ClassObject);
92
93 var args = pi[pi.Length - 1].ParameterType.GetGenericArguments();
94 while (IsPackedArgsContainer(args[args.Length - 1]))
95 {
96 for (int i = 0; i < args.Length - 1; i++)
97 list.Add(context.ClassLoaderFactory.GetJavaTypeFromType(args[i]).ClassObject);
98
99 args = args[args.Length - 1].GetGenericArguments();
100 }
101
102 for (int i = 0; i < args.Length; i++)
103 list.Add(context.ClassLoaderFactory.GetJavaTypeFromType(args[i]).ClassObject);
104
105 types = list.ToArray();
106 }
107 else
108 {
109 types = new java.lang.Class[pi.Length];
110 for (int i = 0; i < types.Length; i++)
111 types[i] = context.ClassLoaderFactory.GetJavaTypeFromType(pi[i].ParameterType).ClassObject;
112 }
113
114 return java.lang.invoke.MethodType.methodType(context.ClassLoaderFactory.GetJavaTypeFromType(mi.ReturnType).ClassObject, types);
115 }

162 DynamicMethodBuilder(RuntimeContext context, string name, java.lang.invoke.MethodType type, Type container, object target, object value, Type owner, bool useBasicTypes)
163 {
164 this.context = context ?? throw new ArgumentNullException(nameof(context));
165 this.type = type;
166 this.container = container;
167
168 delegateType = useBasicTypes ? context.MethodHandleUtil.GetMemberWrapperDelegateType(type) : context.MethodHandleUtil.GetDelegateTypeForInvokeExact(type);
169 firstBoundValue = target;
170 secondBoundValue = value;
171
172 var mi = context.MethodHandleUtil.GetDelegateInvokeMethod(delegateType);
173
174 Type[] paramTypes;
175 if (container != null)
176 {
177 firstArg = 1;
178 paramTypes = GetParameterTypes(container, mi);
179 }
180 else if (target != null)
181 {
182 firstArg = 1;
183 paramTypes = GetParameterTypes(target.GetType(), mi);
184 }
185 else
186 {
187 paramTypes = GetParameterTypes(mi);
188 }
189
190 if (!ReflectUtil.CanOwnDynamicMethod(owner))
191 owner = typeof(DynamicMethodBuilder);
192
193 dm = new DynamicMethod(name, mi.ReturnType, paramTypes, owner, true);
194 ilgen = context.CodeEmitterFactory.Create(dm);
195
196 if (type.parameterCount() > MaxArity)
197 {
198 var pi = mi.GetParameters();
199 packedArgType = pi[pi.Length - 1].ParameterType;
200 packedArgPos = pi.Length - 1 + firstArg;
201 }
202 else
203 {
204 packedArgPos = int.MaxValue;
205 }
206 }

208 internal static Delegate CreateVoidAdapter(RuntimeContext context, global::java.lang.invoke.MethodType type)
209 {
210 var dm = new DynamicMethodBuilder(context, "VoidAdapter", type.changeReturnType(global::java.lang.Void.TYPE), null, null, null, null, true);
211 var targetDelegateType = context.MethodHandleUtil.GetMemberWrapperDelegateType(type);
212 dm.Ldarg(0);
213 dm.EmitCheckcast(context.JavaBase.TypeOfJavaLangInvokeMethodHandle);
214 dm.ilgen.Emit(OpCodes.Ldfld, typeof(global::java.lang.invoke.MethodHandle).GetField(METHOD_HANDLE_FORM_FIELD, BindingFlags.Instance | BindingFlags.NonPublic));
215 dm.ilgen.Emit(OpCodes.Ldfld, typeof(global::java.lang.invoke.LambdaForm).GetField(LAMBDA_FORM_VMENTRY_FIELD, BindingFlags.Instance | BindingFlags.NonPublic));
216 dm.ilgen.Emit(OpCodes.Ldfld, typeof(global::java.lang.invoke.MemberName).GetField(MEMBER_NAME_VMTARGET_FIELD, BindingFlags.Instance | BindingFlags.NonPublic));
217 dm.ilgen.Emit(OpCodes.Castclass, targetDelegateType);
218 for (int i = 0; i < type.parameterCount(); i++)
219 dm.Ldarg(i);
220
221 dm.CallDelegate(targetDelegateType);
222 dm.ilgen.Emit(OpCodes.Pop);
223
224 dm.Ret();
225 return dm.CreateDelegate();
226 }

228 internal static DynamicMethod CreateInvokeExact(RuntimeContext context, global::java.lang.invoke.MethodType type)
229 {
230 FinishTypes(type);
231
232 var dm = new DynamicMethodBuilder(context, "InvokeExact", type, typeof(java.lang.invoke.MethodHandle), null, null, null, false);
233 var targetDelegateType = context.MethodHandleUtil.GetMemberWrapperDelegateType(type.insertParameterTypes(0, context.JavaBase.TypeOfJavaLangInvokeMethodHandle.ClassObject));
234 dm.ilgen.Emit(OpCodes.Ldarg_0);
235 dm.ilgen.Emit(OpCodes.Ldfld, typeof(global::java.lang.invoke.MethodHandle).GetField(METHOD_HANDLE_FORM_FIELD, BindingFlags.Instance | BindingFlags.NonPublic));
236 dm.ilgen.Emit(OpCodes.Ldfld, typeof(global::java.lang.invoke.LambdaForm).GetField(LAMBDA_FORM_VMENTRY_FIELD, BindingFlags.Instance | BindingFlags.NonPublic));
237 if (type.returnType() == java.lang.Void.TYPE)
238 dm.ilgen.Emit(OpCodes.Call, typeof(MethodHandleUtil).GetMethod(nameof(GetVoidAdapter), BindingFlags.Static | BindingFlags.NonPublic));
239 else
240 dm.ilgen.Emit(OpCodes.Ldfld, typeof(java.lang.invoke.MemberName).GetField(MEMBER_NAME_VMTARGET_FIELD, BindingFlags.Instance | BindingFlags.NonPublic));
241
242 dm.ilgen.Emit(OpCodes.Castclass, targetDelegateType);
243 dm.ilgen.Emit(OpCodes.Ldarg_0);
244 for (int i = 0; i < type.parameterCount(); i++)
245 {
246 dm.Ldarg(i);
247 var tw = RuntimeJavaType.FromClass(type.parameterType(i));
248 if (tw.IsNonPrimitiveValueType)
249 tw.EmitBox(dm.ilgen);
250 else if (tw.IsGhost)
251 tw.EmitConvSignatureTypeToStackType(dm.ilgen);
252 else if (tw == context.PrimitiveJavaTypeFactory.BYTE)
253 dm.ilgen.Emit(OpCodes.Conv_I1);
254 }
255 dm.CallDelegate(targetDelegateType);
256
257 var retType = RuntimeJavaType.FromClass(type.returnType());
258 if (retType.IsNonPrimitiveValueType)
259 retType.EmitUnbox(dm.ilgen);
260 else if (retType.IsGhost)
261 retType.EmitConvStackTypeToSignatureType(dm.ilgen, null);
262 else if (!retType.IsPrimitive && retType != context.JavaBase.TypeOfJavaLangObject)
263 dm.EmitCheckcast(retType);
264
265 dm.Ret();
266 dm.ilgen.DoEmit();
267 return dm.dm;
268 }

288 {
289 var type = mn.getMethodType().insertParameterTypes(0, mn.getDeclaringClass());
290 var targetDelegateType = context.MethodHandleUtil.GetMemberWrapperDelegateType(type);
291 var dm = new DynamicMethodBuilder(context, "DirectMethodHandle." + mn.getName() + type, type, typeof(Container<,>).MakeGenericType(typeof(object), typeof(IKVM.Runtime.InvokeCache<>).MakeGenericType(targetDelegateType)), null, null, null, true);
292 dm.Ldarg(0);
293 dm.EmitCheckcast(context.JavaBase.TypeOfJavaLangInvokeMethodHandle);
294 switch (mn.getName())
295 {

312 dm.Ldarg(i);
313 dm.CallDelegate(targetDelegateType);
314
315 dm.Ret();

342 internal static Delegate CreateMemberName(RuntimeContext context, RuntimeJavaMethod mw, global::java.lang.invoke.MethodType type, bool doDispatch)
343 {
344 FinishTypes(type);
345
346 var tw = mw.DeclaringType;
347 var owner = tw.TypeAsBaseType;
348 var dm = new DynamicMethodBuilder(context, "MemberName:" + mw.DeclaringType.Name + "::" + mw.Name + mw.Signature, type, null, mw.HasCallerID ? DynamicCallerIDProvider.Instance : null, null, owner, true);
349 for (int i = 0, count = type.parameterCount(); i < count; i++)
350 {
351 if (i == 0 && !mw.IsStatic && (tw.IsGhost || tw.IsNonPrimitiveValueType || tw.IsRemapped) && (!mw.IsConstructor || tw != context.JavaBase.TypeOfJavaLangString))
352 {
353 if (tw.IsGhost || tw.IsNonPrimitiveValueType)
354 {
355 dm.LoadFirstArgAddress(tw);
356 }
357 else
358 {
359 Debug.Assert(tw.IsRemapped);
360
361 // TODO this must be checked
362 dm.Ldarg(0);
363 if (mw.IsConstructor)
364 dm.EmitCastclass(tw.TypeAsBaseType);
365 else if (tw != context.JavaBase.TypeOfCliSystemObject)
366 dm.EmitCheckcast(tw);
367 }
368 }
369 else
370 {
371 dm.Ldarg(i);
372 var argType = RuntimeJavaType.FromClass(type.parameterType(i));
373 if (!argType.IsPrimitive)
374 {
375 if (argType.IsUnloadable)
376 {
377
378 }
379 else if (argType.IsNonPrimitiveValueType)
380 {
381 dm.Unbox(argType);
382 }
383 else if (argType.IsGhost)
384 {
385 dm.UnboxGhost(argType);
386 }
387 else
388 {
389 dm.EmitCheckcast(argType);
390 }
391 }
392 }
393 }
394
395 if (mw.HasCallerID)
396 dm.LoadCallerID();
397
398 // special case for Object.clone() and Object.finalize()
399 if (mw.IsFinalizeOrClone)
400 {
401 if (doDispatch)
402 {
403 mw.EmitCallvirtReflect(dm.ilgen);
404 }
405 else
406 {
407 // we can re-use the implementations from cli.System.Object (even though the object may not in-fact extend cli.System.Object)
408 context.JavaBase.TypeOfCliSystemObject.GetMethod(mw.Name, mw.Signature, false).EmitCall(dm.ilgen);
409 }
410 }
411 else if (doDispatch && !mw.IsStatic)
412 {
413 dm.Callvirt(mw);
414 }
415 else
416 {
417 dm.Call(mw);
418 }
419
420 var retType = RuntimeJavaType.FromClass(type.returnType());
421 if (retType.IsUnloadable)
422 {
423
424 }
425 else if (retType.IsNonPrimitiveValueType)
426 {
427 dm.Box(retType);
428 }
429 else if (retType.IsGhost)
430 {
431 dm.BoxGhost(retType);
432 }
433 else if (retType == context.PrimitiveJavaTypeFactory.BYTE)
434 {
435 dm.CastByte();
436 }
437
438 dm.Ret();
439 return dm.CreateDelegate();
440 }

519 internal void BoxArgs(int start)
520 {
521 int paramCount = type.parameterCount();
522 ilgen.EmitLdc_I4(paramCount - start);
