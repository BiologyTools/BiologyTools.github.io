158 static short[] ConvertByteArrayToShortArray(byte[] byteArray, bool littleEndian)
159 {
160 if (byteArray.Length % 2 != 0)
161 {
162 throw new ArgumentException("The length of the byte array must be even.");
163 }
164
165 // Each short is made from two bytes
166 short[] shortArray = new short[byteArray.Length / 2];
167
168 for (int i = 0; i < shortArray.Length; i++)
169 {
170 if (littleEndian)
171 {
172 // If the system is little-endian, reverse the byte order
173 shortArray[i] = (short)((byteArray[i * 2 + 1] << 8) | byteArray[i * 2]);
174 }
175 else
176 {
177 // If the system is big-endian, keep the byte order as is
178 shortArray[i] = (short)((byteArray[i * 2] << 8) | byteArray[i * 2 + 1]);
179 }
180 }
181 return shortArray;
182 }
183 public static ImagePlus GetImagePlus(BioImage bm)
184 {
185 ImageStack ims = new ImageStack(bm.SizeX, bm.SizeY);
186 if (bm.Buffers[0].PixelFormat == AForge.PixelFormat.Format16bppGrayScale)
187 {
188 // Populate the stack with 16-bit images
189 for (int t = 0; t < bm.SizeT; t++)
190 {
191 for (int z = 0; z < bm.SizeZ; z++)
192 {
193 for (int c = 0; c < bm.SizeC; c++)
194 {
195 short[] pixels = ConvertByteArrayToShortArray(bm.Buffers[bm.GetFrameIndex(z, c, t)].Bytes, true);
196 ImageProcessor ip = new ShortProcessor(bm.SizeX, bm.SizeY, pixels, java.awt.image.ColorModel.getRGBdefault());
197 ims.addSlice("Z" + z + "C" + c + "T" + t, ip);
198 }
199 }
200 }
201 }
202 else if (bm.Buffers[0].PixelFormat == AForge.PixelFormat.Format8bppIndexed)
203 {
204 // Create a grayscale IndexColorModel
205 byte[] grayscale = new byte[256];
206 for (int i = 0; i < 256; i++)
207 {
208 grayscale[i] = (byte)i;
209 }
210 IndexColorModel colorModel = new IndexColorModel(8, 256, grayscale, grayscale, grayscale);
211 for (int t = 0; t < bm.SizeT; t++)
212 {
213 for (int z = 0; z < bm.SizeZ; z++)
214 {
215 for (int c = 0; c < bm.SizeC; c++)
216 {
217 ImageProcessor ip = new ByteProcessor(bm.SizeX, bm.SizeY, bm.Buffers[bm.GetFrameIndex(z, c, t)].Bytes, colorModel);
218 ims.addSlice("Z" + z + "C" + c + "T" + t, ip);
219 }
220 }
221 }
222 }
223 else if (bm.Buffers[0].PixelFormat == AForge.PixelFormat.Format24bppRgb)
224 {
225 for (int t = 0; t < bm.SizeT; t++)
226 {
227 for (int z = 0; z < bm.SizeZ; z++)
228 {
229 for (int c = 0; c < bm.SizeC; c++)
230 {
231 byte[] bts = bm.Buffers[bm.GetFrameIndex(z, c, t)].Bytes;
232 // Convert the byte array to an int array for the RGB processor
233 int[] rgbPixels = new int[bm.SizeX * bm.SizeY];
234 for (int i = 0; i < rgbPixels.Length; i++)
235 {
236 int r = bts[i * 3] & 0xFF; // Red
237 int g = bts[i * 3 + 1] & 0xFF; // Green
238 int b = bts[i * 3 + 2] & 0xFF; // Blue
239 rgbPixels[i] = (r << 16) | (g << 8) | b; // Combine into RGB int
240 }
241 // Create a ColorProcessor
242 ImageProcessor ip = new ColorProcessor(bm.SizeX, bm.SizeY, rgbPixels);
243 ims.addSlice("Z" + z + "C" + c + "T" + t, ip);
244 }
245 }
246 }
247 }
248 else if (bm.Buffers[0].PixelFormat == AForge.PixelFormat.Format32bppArgb || bm.Buffers[0].PixelFormat == AForge.PixelFormat.Format32bppRgb)
249 {
250 for (int t = 0; t < bm.SizeT; t++)
251 {
252 for (int z = 0; z < bm.SizeZ; z++)
253 {
254 for (int c = 0; c < bm.SizeC; c++)
255 {
256 byte[] bts = bm.Buffers[bm.GetFrameIndex(z, c, t)].Bytes;
257 // Convert the byte array to an int array for the ColorProcessor
258 int[] rgbaPixels = new int[bm.SizeX * bm.SizeY];
259 for (int i = 0; i < rgbaPixels.Length; i++)
260 {
261 int r = bts[i * 4] & 0xFF; // Red
262 int g = bts[i * 4 + 1] & 0xFF; // Green
263 int b = bts[i * 4 + 2] & 0xFF; // Blue
264 int a = bts[i * 4 + 3] & 0xFF; // Alpha
265 rgbaPixels[i] = (a << 24) | (r << 16) | (g << 8) | b; // Combine into ARGB int
266 }
267 // Create a ColorProcessor with the ARGB int array
268 ImageProcessor ip = new ColorProcessor(bm.SizeX, bm.SizeY, rgbaPixels);
269 ims.addSlice("Z" + z + "C" + c + "T" + t, ip);
270 }
271 }
272 }
273 }
274 else if (bm.Buffers[0].PixelFormat == AForge.PixelFormat.Format48bppRgb)
275 {
276 bm.To16Bit();
277 // Populate the stack with 16-bit images
278 for (int t = 0; t < bm.SizeT; t++)
279 {
280 for (int z = 0; z < bm.SizeZ; z++)
281 {
282 for (int c = 0; c < bm.SizeC; c++)
283 {
284 short[] pixels = ConvertByteArrayToShortArray(bm.Buffers[bm.GetFrameIndex(z, c, t)].Bytes, true);
285 ImageProcessor ip = new ShortProcessor(bm.SizeX, bm.SizeY, pixels, java.awt.image.ColorModel.getRGBdefault());
286 ims.addSlice("Z" + z + "C" + c + "T" + t, ip);
287 }
288 }
289 }
290 }
291 ImagePlus imp = new ImagePlus(bm.ID, ims);
292 imp.setDimensions(bm.SizeC, bm.SizeZ, bm.SizeT);
293 return imp;

321 public static BioImage GetBioImage(ImagePlus pl, VolumeD vol, double PhysicalX, double PhysicalY, double PhysicalZ)
322 {
323 BioImage bm = new BioImage(pl.getTitle());
324 ImageStack st = pl.getImageStack();
325 int b = pl.getBitDepth();
326 int slices = pl.getNSlices();
327 int chs = pl.getNChannels();
328 int frs = pl.getNFrames();
329 int rgb = (int)(pl.getBytesPerPixel() * 8);
330 bool isrgb = isRGB(pl);
331 bm.StackOrder = BioImage.Order.ZCT;
332 bm.UpdateCoords(slices, chs, frs, BioImage.Order.ZCT);
333 AForge.Bitmap[] outputBuffers = new AForge.Bitmap[Math.Max(1, slices * chs * frs)];
334 for (int t = 0; t < frs; t++)
335 {
336 for (int z = 0; z < slices; z++)
337 {
338 for (int c = 0; c < chs; c++)
339 {
340 AForge.Bitmap bmp;
341 if (!isrgb)
342 {
343 if (rgb > 8)
344 {
345 bmp = new AForge.Bitmap(st.getWidth(), st.getHeight(), AForge.PixelFormat.Format16bppGrayScale);
346 }
347 else
348 {
349 bmp = new AForge.Bitmap(st.getWidth(), st.getHeight(), AForge.PixelFormat.Format8bppIndexed);
350 }
351 }
352 else
353 {
354 bmp = new AForge.Bitmap(st.getWidth(), st.getHeight(), AForge.PixelFormat.Format24bppRgb);
355 }
356 int ind = bm.GetFrameIndex(z, c, t);
357 int stackIndex = GetHyperStackSliceIndex(pl, z, c, t);
358 outputBuffers[ind] = bmp;
359 if (!isrgb)
360 {
361 for (int y = 0; y < pl.getHeight(); y++)
362 {
363 for (int x = 0; x < pl.getWidth(); x++)
364 {
365 double d = st.getVoxel(x, y, stackIndex);
366 if (bmp.PixelFormat == PixelFormat.Format16bppGrayScale)
367 bmp.SetValue(x, y, (ushort)d);
368 else if (bmp.PixelFormat == PixelFormat.Format8bppIndexed)
369 bmp.SetValue(x, y, (byte)d);
370 }
371 }
372 }
373 else
374 {
375 ImageProcessor ip = st.getProcessor(stackIndex + 1); // ImageStack uses 1-based indexing
376 // Cast to ColorProcessor to access RGB data
377 ColorProcessor colorProcessor = (ColorProcessor)ip;
378 for (int y = 0; y < pl.getHeight(); y++)
379 {
380 for (int x = 0; x < pl.getWidth(); x++)
381 {
382 int v = colorProcessor.getPixel(x, y);
383 int rv = (v >> 16) & 0xFF; // Red
384 int gv = (v >> 8) & 0xFF; // Green
385 int bv = v & 0xFF; // Blue
386 bmp.SetValue(x, y, 0, (byte)rv);
387 bmp.SetValue(x, y, 1, (byte)gv);
388 bmp.SetValue(x, y, 2, (byte)bv);
389 }
390 }
391
392 }
393 }
394 }
395 }
396 bm.Buffers.AddRange(outputBuffers.Where(buf => buf != null));
397 for (int t = 0; t < chs; t++)
398 {
399 if (isrgb)
400 bm.Channels.Add(new AForge.Channel(t, rgb, 3));
401 else
402 bm.Channels.Add(new AForge.Channel(t, rgb, 1));
403 if (t == 0)
404 {
405 bm.rgbChannels[0] = 0;
406 }
407 else
408 if (t == 1)
409 {
410 bm.rgbChannels[1] = 1;
411 }
412 else
413 if (t == 2)
414 {
415 bm.rgbChannels[2] = 2;
416 }
417 }
418 bm.Resolutions.Add(new Resolution(pl.getWidth(), pl.getHeight(), bm.Buffers[0].PixelFormat, PhysicalX, PhysicalY, PhysicalZ, vol.Location.X, vol.Location.Y, vol.Location.X));
419 bm.Volume = vol;
420 bm.littleEndian = BitConverter.IsLittleEndian;
421 bm.seriesCount = 1;
422 BioImage.AutoThreshold(bm, true);
423 if (rgb > 8)
424 bm.StackThreshold(true);
425 else
426 bm.StackThreshold(false);
427
428 if (bm.RGBChannelCount == 4)
429 {
430 bm.Channels.Last().SamplesPerPixel = 4;
431 }
432 return bm;
433 }

798 private static bool TryExtractBareRunCommand(string con, out string commandName)
799 {
800 commandName = null;
801 if (string.IsNullOrWhiteSpace(con))
802 return false;
803
804 string trimmed = con.Trim();
805 const string prefix = "run(\"";
806 const string suffix = "\");";
807 if (!trimmed.StartsWith(prefix, StringComparison.Ordinal) || !trimmed.EndsWith(suffix, StringComparison.Ordinal))
808 return false;
809
810 string body = trimmed.Substring(prefix.Length, trimmed.Length - prefix.Length - suffix.Length);
811 if (body.Contains("\",", StringComparison.Ordinal))
812 return false;
813
814 commandName = body;
815 return !string.IsNullOrWhiteSpace(commandName);

835
836 string candidatePrefix = $"run(\"{safeCommand}\"";
837 string[] lines = delta.Split(new[] { "\r\n", "\n" }, StringSplitOptions.RemoveEmptyEntries);

841 if (!line.StartsWith(candidatePrefix, StringComparison.Ordinal) || !line.EndsWith(");", StringComparison.Ordinal))

845 return false;

848 return false;
