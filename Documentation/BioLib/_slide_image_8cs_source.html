27 private static void GetRegionDimensions(RegionResult region, int fallbackWidth, int fallbackHeight, out int width, out int height)
28 {
29 width = Math.Max(1, fallbackWidth);
30 height = Math.Max(1, fallbackHeight);
31
32 if (region?.Shape == null || region.Shape.Length == 0)
33 return;
34
35 if (region.Axes != null && region.Axes.Length == region.Shape.Length)
36 {
37 for (int i = 0; i < region.Axes.Length; i++)
38 {
39 string axisName = region.Axes[i].Name?.ToLowerInvariant() ?? string.Empty;
40 if (axisName == "x")
41 width = (int)Math.Max(1, region.Shape[i]);
42 else if (axisName == "y")
43 height = (int)Math.Max(1, region.Shape[i]);
44 }
45 return;
46 }
47
48 if (region.Shape.Length >= 2)
49 {
50 width = (int)Math.Max(1, region.Shape[^1]);
51 height = (int)Math.Max(1, region.Shape[^2]);
52 }
53 }

208 public static double CalculateBaseFactor(int originalResolution, int lastLevelResolution, int totalLevels)
209 {
210 if (totalLevels <= 1)
211 {
212 throw new ArgumentException("Total levels must be greater than 1 to calculate a base factor.");
213 }
214 if (lastLevelResolution <= 0 || originalResolution <= 0)
215 {
216 throw new ArgumentException("Resolutions must be greater than 0.");
217 }
218
219 // Calculate the base downsampling factor
220 double baseFactor = Math.Pow((double)originalResolution / lastLevelResolution, 1.0 / (totalLevels - 1));
221 return baseFactor;
