247 private static string ResolveExportedImagePath(string exportPath)
248 {
249 if (File.Exists(exportPath))
250 return exportPath.Replace("\\", "/");
251
252 string dir = Path.GetDirectoryName(exportPath);
253 if (string.IsNullOrEmpty(dir) || !Directory.Exists(dir))
254 return null;
255
256 string stem = Path.GetFileNameWithoutExtension(exportPath);
257 string[] candidates = Directory.GetFiles(dir)
258 .Where(f =>
259 {
260 string name = Path.GetFileName(f);
261 return name.StartsWith(stem, StringComparison.OrdinalIgnoreCase) &&
262 (name.EndsWith(".tif", StringComparison.OrdinalIgnoreCase) ||
263 name.EndsWith(".tiff", StringComparison.OrdinalIgnoreCase));
264 })
265 .OrderByDescending(File.GetLastWriteTimeUtc)
266 .ToArray();
267
268 return candidates.Length > 0 ? candidates[0].Replace("\\", "/") : null;
269 }

272 private static bool TryGetPyramidLevelGeometry(BioImage b, int level, out Resolution res, out int width, out int height, out double unitsPerPixel)
273 {
274 res = default;
275 width = 0;
276 height = 0;
277 unitsPerPixel = 0;
278
279 if (b == null || b.Resolutions.Count == 0)
280 return false;
281
282 level = Math.Clamp(level, 0, b.Resolutions.Count - 1);
283 res = b.Resolutions[level];
284 int schemaWidth = 0;
285 int schemaHeight = 0;
286 unitsPerPixel = res.PhysicalSizeX;
287
288 if (b.OpenSlideBase?.Schema?.Resolutions != null && b.OpenSlideBase.Schema.Resolutions.Count > level)
289 {
290 var schemaRes = b.OpenSlideBase.Schema.Resolutions[level];
291 schemaWidth = GetIntProperty(schemaRes, "Width", "SizeX");
292 schemaHeight = GetIntProperty(schemaRes, "Height", "SizeY");
293 unitsPerPixel = schemaRes.UnitsPerPixel;
294 }
295 else if (b.SlideBase?.Schema?.Resolutions != null && b.SlideBase.Schema.Resolutions.Count > level)
296 {
297 var schemaRes = b.SlideBase.Schema.Resolutions[level];
298 schemaWidth = GetIntProperty(schemaRes, "Width", "SizeX");
299 schemaHeight = GetIntProperty(schemaRes, "Height", "SizeY");
300 unitsPerPixel = schemaRes.UnitsPerPixel;
301 }
302
303 width = schemaWidth > 0 ? schemaWidth : (res.SizeX > 0 ? res.SizeX : b.SizeX);
304 height = schemaHeight > 0 ? schemaHeight : (res.SizeY > 0 ? res.SizeY : b.SizeY);
305 return width > 0 && height > 0;
306 }

318 private static PyramidExtractionRequest BuildExtractionRequest(BioImage b, int level, bool currentViewOnly)
319 {
320 if (!TryGetPyramidLevelGeometry(b, level, out Resolution res, out int levelWidth, out int levelHeight, out double schemaUnitsPerPixel))
321 return null;
322
323 double physX = res.PhysicalSizeX > 0 ? res.PhysicalSizeX : Math.Max(schemaUnitsPerPixel, 1.0);
324 double physY = res.PhysicalSizeY > 0 ? res.PhysicalSizeY : physX;
325
326 var request = new PyramidExtractionRequest
327 {
328 Level = Math.Clamp(level, 0, b.Resolutions.Count - 1),
329 X = 0,
330 Y = 0,
331 Width = levelWidth,
332 Height = levelHeight,
333 StageX = res.StageSizeX,
334 StageY = res.StageSizeY,
335 IsCurrentViewOnly = false,
336 Description = $"pyramid level {level + 1}/{b.Resolutions.Count}"
337 };
338
339 if (!currentViewOnly)
340 return request;
341
342 if (b.PyramidalSize.Width <= 1 || b.PyramidalSize.Height <= 1)
343 return request;
344
345 double viewUnitsPerPixel = b.Resolution > 0 ? b.Resolution : physX;
346 double viewWorldX = b.PyramidalOrigin.X;
347 double viewWorldY = b.PyramidalOrigin.Y;
348 double viewWorldW = Math.Max(1, b.PyramidalSize.Width) * viewUnitsPerPixel;
349 double viewWorldH = Math.Max(1, b.PyramidalSize.Height) * (b.Resolution > 0 ? b.Resolution : physY);
350
351 int x = (int)Math.Floor((viewWorldX - res.StageSizeX) / physX);
352 int y = (int)Math.Floor((viewWorldY - res.StageSizeY) / physY);
353 int w = (int)Math.Ceiling(viewWorldW / physX);
354 int h = (int)Math.Ceiling(viewWorldH / physY);
355
356 x = Math.Max(0, x);
357 y = Math.Max(0, y);
358 w = Math.Min(levelWidth - x, Math.Max(1, w));
359 h = Math.Min(levelHeight - y, Math.Max(1, h));
360
361 if (w <= 0 || h <= 0)
362 return request;
363
364 long fullArea = Math.Max(1L, (long)levelWidth * levelHeight);
365 long regionArea = Math.Max(1L, (long)w * h);
366 if (regionArea * 10 >= fullArea * 9)
367 return request;
368
369 request.X = x;
370 request.Y = y;
371 request.Width = w;
372 request.Height = h;
373 request.StageX = res.StageSizeX + (x * physX);
374 request.StageY = res.StageSizeY + (y * physY);
375 request.IsCurrentViewOnly = true;
376 request.Description = $"visible region of level {level + 1}/{b.Resolutions.Count}";
377 return request;
378 }
379 private static string GetExtractionCacheKey(BioImage source, PyramidExtractionRequest request)
380 {
381 string raw = string.Join("|",
382 source?.SourceFile ?? source?.ID ?? string.Empty,
383 request.Level,
384 request.X,
385 request.Y,
386 request.Width,
387 request.Height,
388 source?.SizeZ ?? 0,
389 source?.SizeC ?? 0,
390 source?.SizeT ?? 0);
391
392 byte[] hash = SHA256.HashData(Encoding.UTF8.GetBytes(raw));
393 StringBuilder sb = new StringBuilder(hash.Length * 2);
394 for (int i = 0; i < hash.Length; i++)
395 sb.Append(hash[i].ToString("x2"));
396 return sb.ToString();
397 }
398 private static BioImage TryGetCachedExtraction(string cacheKey)
399 {
400 if (string.IsNullOrWhiteSpace(cacheKey))
401 return null;
402
403 lock (extractionCacheLock)
404 {
405 if (!extractionCache.TryGetValue(cacheKey, out BioImage cached) || cached == null)
406 return null;
407
408 var node = extractionCacheOrder.Find(cacheKey);
409 if (node != null)
410 {
411 extractionCacheOrder.Remove(node);
412 extractionCacheOrder.AddLast(node);
413 }
414 return cached;
415 }
416 }
417 private static void StoreCachedExtraction(string cacheKey, BioImage image)
418 {
419 if (string.IsNullOrWhiteSpace(cacheKey) || image == null)
420 return;
421
422 lock (extractionCacheLock)
423 {
424 if (extractionCache.ContainsKey(cacheKey))
425 return;
426
427 extractionCache[cacheKey] = image;
428 extractionCacheOrder.AddLast(cacheKey);
429 while (extractionCacheOrder.Count > MaxExtractionCacheEntries)
430 {
431 string evictedKey = extractionCacheOrder.First.Value;
432 extractionCacheOrder.RemoveFirst();
433 if (extractionCache.TryGetValue(evictedKey, out BioImage evicted) && evicted != null)
434 {
435 try
436 {
437 evicted.Dispose();
438 }
439 catch
440 {
441 }
442 }
443 extractionCache.Remove(evictedKey);
444 }
445 }

451 private static async Task<BioImage> PreparePyramidalLevelImage(BioImage b, int level, bool currentViewOnly)
452 {
453 if (b == null)
454 return null;
455 if (!b.isPyramidal || b.Resolutions.Count == 0)
456 return b;
457
458 level = Math.Clamp(level, 0, b.Resolutions.Count - 1);
459 PyramidExtractionRequest request = BuildExtractionRequest(b, level, currentViewOnly);
460 if (request == null)
461 return null;
462
463 string cacheKey = IsRemotePyramidalSource(b) ? GetExtractionCacheKey(b, request) : null;
464 if (!string.IsNullOrWhiteSpace(cacheKey))
465 {
466 BioImage cached = TryGetCachedExtraction(cacheKey);
467 if (cached != null)
468 {
469 ReportProgress(100, $"Using cached {request.Description}");
470 return cached;
471 }
472 }
473
474 Console.WriteLine($"[ImageJ.PreparePyramidalLevelImage] extracting {request.Description} from {b.ID}");
475 BioImage.Status = $"Extracting {request.Description}";
476
477 if (!TryGetPyramidLevelGeometry(b, level, out Resolution res, out _, out _, out double schemaUnitsPerPixel))
478 return null;
479
480 BioImage pyrlevel = BioImage.CopyInfo(b, true, true);
481 Resolution fixedRes = pyrlevel.Resolutions[level];
482 fixedRes.SizeX = request.Width;
483 fixedRes.SizeY = request.Height;
484 fixedRes.PhysicalSizeX = res.PhysicalSizeX;
485 fixedRes.PhysicalSizeY = res.PhysicalSizeY;
486 fixedRes.PhysicalSizeZ = res.PhysicalSizeZ;
487 fixedRes.StageSizeX = request.StageX;
488 fixedRes.StageSizeY = request.StageY;
489 fixedRes.StageSizeZ = res.StageSizeZ;
490 pyrlevel.Resolutions.Clear();
491 pyrlevel.Resolutions.Add(fixedRes);
492
493 pyrlevel.PyramidalOrigin = new PointD(0, 0);
494 pyrlevel.PyramidalSize = new AForge.Size(request.Width, request.Height);
495 pyrlevel.Resolution = schemaUnitsPerPixel > 0 ? schemaUnitsPerPixel : fixedRes.PhysicalSizeX;
496 pyrlevel.Level = 0;
497 pyrlevel.StackOrder = BioImage.Order.ZCT;
498 pyrlevel.UpdateCoords(b.SizeZ, b.SizeC, b.SizeT, BioImage.Order.ZCT);
499 pyrlevel.Volume = new VolumeD(
500 new Point3D(request.StageX, request.StageY, fixedRes.StageSizeZ),
501 new Point3D(request.Width * fixedRes.PhysicalSizeX, request.Height * fixedRes.PhysicalSizeY, Math.Max(1, b.SizeZ) * fixedRes.PhysicalSizeZ));
502
503 pyrlevel.Buffers.Clear();
504 var planes = new List<(int Sequence, int Index, int Z, int C, int T)>(Math.Max(1, b.SizeT * b.SizeC * b.SizeZ));
505 for (int t = 0; t < b.SizeT; t++)
506 {
507 for (int c = 0; c < b.SizeC; c++)
508 {
509 for (int z = 0; z < b.SizeZ; z++)
510 {
511 int index = b.GetFrameIndex(z, c, t);
512 if (index < 0)
513 continue;
514
515 planes.Add((planes.Count, index, z, c, t));
516 }
517 }
518 }
519
520 int totalPlanes = planes.Count;
521 if (totalPlanes == 0)
522 return null;
523
524 Bitmap[] extractedPlanes = new Bitmap[totalPlanes];
525 int loadedPlanes = 0;
526 int parallelism = GetExtractionParallelism(b);
527 using (SemaphoreSlim gate = new SemaphoreSlim(parallelism, parallelism))
528 {
529 Task[] extractionTasks = planes.Select(async plane =>
530 {
531 await gate.WaitAsync().ConfigureAwait(false);
532 try
533 {
534 extractedPlanes[plane.Sequence] = await b.GetTile(
535 plane.Index, level,
536 request.X, request.Y, request.Width, request.Height,
537 new AForge.ZCT(plane.Z, plane.C, plane.T),
538 true).ConfigureAwait(false);
539
540 int completedPlanes = Interlocked.Increment(ref loadedPlanes);
541 ReportProgress(
542 (completedPlanes * 100f) / totalPlanes,
543 $"Extracting {request.Description} plane T{plane.T + 1}/{b.SizeT} Z{plane.Z + 1}/{b.SizeZ} C{plane.C + 1}/{b.SizeC}");
544 }
545 finally
546 {
547 gate.Release();
548 }
549 }).ToArray();
550
551 await Task.WhenAll(extractionTasks).ConfigureAwait(false);
552 }
553
554 foreach (Bitmap tile in extractedPlanes)
555 {
556 if (tile != null)
557 pyrlevel.Buffers.Add(tile);
558 }
559
560 if (pyrlevel.Buffers.Count == 0)
561 return null;
562
563 if (!string.IsNullOrWhiteSpace(cacheKey))
564 StoreCachedExtraction(cacheKey, pyrlevel);
565 return pyrlevel;
566 }

568 private static int GetIntProperty(object obj, params string[] names)
569 {
570 if (obj == null)
571 return 0;
572
573 foreach (string name in names)
574 {
575 var prop = obj.GetType().GetProperty(name);
576 if (prop == null)
577 continue;
578
579 object value = prop.GetValue(obj);
580 if (value == null)
581 continue;
582
583 if (value is int i)
584 return i;
585
586 if (int.TryParse(value.ToString(), out int parsed))
587 return parsed;
588 }
589
590 return 0;
591 }

700 bm.To48Bit();
701 }

703 outRes.PixelFormat = srcRes.PixelFormat;
704 outRes.PhysicalSizeX = srcRes.PhysicalSizeX;
705 outRes.PhysicalSizeY = srcRes.PhysicalSizeY;
706 outRes.PhysicalSizeZ = srcRes.PhysicalSizeZ;

861 private static (int tileW, int tileH) GetLargestSafeTileSize(int width, int height, int bytesPerSample, int planeCount)
862 {
863 const long maxTileBytes = 128L * 1024L * 1024L;
864 if (width <= 0 || height <= 0)
865 return (1, 1);
866
867 long bytesPerPixelAllPlanes = Math.Max(1L, (long)bytesPerSample * planeCount);
868
869 (int w, int h) best = (1, 1);
870 long bestArea = 1;
871
872 void Consider(int candidateW, int candidateH)
873 {
874 candidateW = Math.Max(1, Math.Min(width, candidateW));
875 candidateH = Math.Max(1, Math.Min(height, candidateH));
876 long bytes = (long)candidateW * candidateH * bytesPerPixelAllPlanes;
877 if (bytes > maxTileBytes)
878 return;
879
880 long area = (long)candidateW * candidateH;
881 if (area > bestArea)
882 {
883 best = (candidateW, candidateH);
884 bestArea = area;
885 }
886 }
887
888 long maxHeightForFullWidth = maxTileBytes / Math.Max(1L, (long)width * bytesPerPixelAllPlanes);
889 if (maxHeightForFullWidth > 0)
890 Consider(width, (int)Math.Min(height, maxHeightForFullWidth));
891
892 long maxWidthForFullHeight = maxTileBytes / Math.Max(1L, (long)height * bytesPerPixelAllPlanes);
893 if (maxWidthForFullHeight > 0)
894 Consider((int)Math.Min(width, maxWidthForFullHeight), height);
895
896 long pixelsBudget = Math.Max(1L, maxTileBytes / bytesPerPixelAllPlanes);
897 int square = Math.Max(1, (int)Math.Min(Math.Min(width, height), Math.Sqrt(pixelsBudget)));
898 for (int delta = 0; delta < 8; delta++)
899 {
900 int candidateW = Math.Max(1, Math.Min(width, square + delta));
901 int candidateH = Math.Max(1, (int)Math.Min(height, pixelsBudget / candidateW));
902 Consider(candidateW, candidateH);
903 }
904
905 if (bestArea > 1)
906 return best;
907
908 return (1, Math.Max(1, (int)Math.Min(height, pixelsBudget)));
909 }
