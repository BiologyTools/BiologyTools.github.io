360 private static void ProgressUpdate()
361 {
362 do
363 {
364 progress.Text = title;
365 progress.Status = BioImage.Status;
366 progress.ProgressValue = BioImage.Progress;
367 Thread.Sleep(250);
368 } while (!done);
369 }
370 private static void StartProgress(string titl, string status)
371 {
372 done = false;
373 title = titl;
374 BioImage.Progress = 0;
375 progress.Status = status;
376 progress.Text = title;
377 if (BioImage.Status == "" || BioImage.Status == null)
378 progress.Status = status;
379 else
380 progress.Status = BioImage.Status;
381 progress.ProgressValue = BioImage.Progress;
382 progress.Show();
383 Application.DoEvents();
384 Thread th = new Thread(ProgressUpdate);
385 th.Start();
386 }
387 private static void StopProgress()
388 {
389 done = true;
390 progress.Hide();

1009 private void TabsView_KeyDown(object sender, KeyEventArgs e)
1010 {
1011 if (Viewer == null)
1012 return;
1013 double moveAmount = 1500 * (1 / App.viewer.Scale.Width);
1014 if (e.KeyCode == Keys.C && e.Control)
1015 {
1016 Viewer.CopySelection();
1017 return;
1018 }
1019 if (e.KeyCode == Keys.V && e.Control)
1020 {
1021 Viewer.PasteSelection();
1022 return;
1023 }
1024 if (e.KeyCode == Keys.Subtract || e.KeyCode == Keys.NumPad7)
1025 {
1026 App.viewer.Scale = new AForge.SizeF(App.viewer.Scale.Width - 0.1f, App.viewer.Scale.Height - 0.1f);
1027 }
1028 if (e.KeyCode == Keys.Add || e.KeyCode == Keys.NumPad9)
1029 {
1030 App.viewer.Scale = new AForge.SizeF(App.viewer.Scale.Width - 0.1f, App.viewer.Scale.Height - 0.1f);
1031 }
1032 if (e.KeyCode == Keys.W || e.KeyCode == Keys.NumPad8)
1033 {
1034 Viewer.Origin = new PointD(Viewer.Origin.X, Viewer.Origin.Y - moveAmount);
1035 }
1036 if (e.KeyCode == Keys.S || e.KeyCode == Keys.NumPad2)
1037 {
1038 Viewer.Origin = new PointD(Viewer.Origin.X, Viewer.Origin.Y + moveAmount);
1039 }
1040 if (e.KeyCode == Keys.A || e.KeyCode == Keys.NumPad4)
1041 {
1042 Viewer.Origin = new PointD(Viewer.Origin.X - moveAmount, Viewer.Origin.Y);
1043 }
1044 if (e.KeyCode == Keys.D || e.KeyCode == Keys.NumPad6)
1045 {
1046 Viewer.Origin = new PointD(Viewer.Origin.X + moveAmount, Viewer.Origin.Y);
1047 }
1048 if (e.KeyCode == Keys.Up)
1049 Microscope.MoveUp(Microscope.viewSize.Y / 2);
1050 if (e.KeyCode == Keys.Down)
1051 Microscope.MoveDown(Microscope.viewSize.Y / 2);
1052 if (e.KeyCode == Keys.Left)
1053 Microscope.MoveRight(-(Microscope.viewSize.X / 2));
1054 if (e.KeyCode == Keys.Right)
1055 Microscope.MoveRight((Microscope.viewSize.X / 2) / 2);
1056 Viewer.UpdateStatus();
1057 Viewer.UpdateView();
